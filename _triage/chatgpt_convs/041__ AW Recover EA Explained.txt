# EA AW Recover Style สำหรับ MQL5

ผมจะเขียน EA ที่ใช้หลักการแบบ AW Recover ให้นะครับ ซึ่งเป็นระบบ Grid Averaging แบบมีการคำนวณ Lot ที่ซับซ้อนเพื่อ Recover ขาดทุน

```mql5
//+------------------------------------------------------------------+
//|                                              AW_Recover_EA.mq5   |
//|                                   Quant Trader - Forex Recovery  |
//+------------------------------------------------------------------+
#property copyright "Quant Trader"
#property version   "1.00"
#property strict

//--- Input Parameters
input group "===== Trading Settings ====="
input ENUM_ORDER_TYPE OrderDirection = ORDER_TYPE_BUY;  // Direction (BUY/SELL)
input double FirstLot = 0.01;                            // First Order Lot Size
input double LotMultiplier = 1.5;                        // Lot Multiplier
input int StepPoints = 300;                              // Step Distance (Points)
input int MaxOrders = 10;                                // Maximum Orders
input double TakeProfitMoney = 50;                       // Target Profit ($)

input group "===== Recovery Settings ====="
input bool UseAWFormula = true;                          // Use AW Recovery Formula
input double RecoveryFactor = 1.3;                       // Recovery Factor

input group "===== Risk Management ====="
input double MaxLotSize = 10.0;                          // Maximum Lot Size
input double MaxDrawdownMoney = 1000;                    // Max Drawdown ($)
input bool UseTrailingStop = false;                      // Use Trailing Stop

input group "===== Other Settings ====="
input int MagicNumber = 88888;                           // Magic Number
input int Slippage = 30;                                 // Slippage (Points)
input string TradeComment = "AW_Recover";                // Trade Comment

//--- Global Variables
int totalOrders = 0;
double lastPrice = 0;
double averagePrice = 0;
double totalLots = 0;
double totalProfit = 0;
bool isTrading = false;
datetime lastBarTime = 0;

//+------------------------------------------------------------------+
//| Expert initialization function                                    |
//+------------------------------------------------------------------+
int OnInit()
{
   Print("=== AW Recover EA Initialized ===");
   Print("Symbol: ", _Symbol);
   Print("First Lot: ", FirstLot);
   Print("Step Points: ", StepPoints);
   Print("Max Orders: ", MaxOrders);
   Print("Target Profit: $", TakeProfitMoney);
   
   return(INIT_SUCCEEDED);
}

//+------------------------------------------------------------------+
//| Expert deinitialization function                                  |
//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
   Print("=== AW Recover EA Stopped ===");
}

//+------------------------------------------------------------------+
//| Expert tick function                                              |
//+------------------------------------------------------------------+
void OnTick()
{
   //--- Update order info
   UpdateOrderInfo();
   
   //--- Check if we need to close all orders (target reached)
   if(totalOrders > 0)
   {
      if(totalProfit >= TakeProfitMoney)
      {
         CloseAllOrders();
         Print("*** TARGET PROFIT REACHED: $", totalProfit, " ***");
         return;
      }
      
      //--- Check max drawdown
      if(MaxDrawdownMoney > 0 && totalProfit <= -MaxDrawdownMoney)
      {
         CloseAllOrders();
         Print("*** MAX DRAWDOWN REACHED: $", totalProfit, " ***");
         return;
      }
   }
   
   //--- Open first order if no orders exist
   if(totalOrders == 0)
   {
      OpenFirstOrder();
      return;
   }
   
   //--- Check if we need to open next order
   if(totalOrders < MaxOrders)
   {
      CheckAndOpenNextOrder();
   }
}

//+------------------------------------------------------------------+
//| Update Order Information                                          |
//+------------------------------------------------------------------+
void UpdateOrderInfo()
{
   totalOrders = 0;
   totalLots = 0;
   totalProfit = 0;
   double sumPrice = 0;
   lastPrice = 0;
   
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol)
      {
         if(PositionGetInteger(POSITION_MAGIC) == MagicNumber)
         {
            totalOrders++;
            double lots = PositionGetDouble(POSITION_VOLUME);
            double openPrice = PositionGetDouble(POSITION_PRICE_OPEN);
            double profit = PositionGetDouble(POSITION_PROFIT);
            double swap = PositionGetDouble(POSITION_SWAP);
            double commission = PositionGetDouble(POSITION_COMMISSION);
            
            totalLots += lots;
            sumPrice += openPrice * lots;
            totalProfit += profit + swap + commission;
            
            //--- Keep track of last order price
            if(totalOrders == 1 || (OrderDirection == ORDER_TYPE_BUY && openPrice < lastPrice) || 
               (OrderDirection == ORDER_TYPE_SELL && openPrice > lastPrice) || lastPrice == 0)
            {
               lastPrice = openPrice;
            }
         }
      }
   }
   
   //--- Calculate average price
   if(totalLots > 0)
      averagePrice = sumPrice / totalLots;
   else
      averagePrice = 0;
}

//+------------------------------------------------------------------+
//| Open First Order                                                  |
//+------------------------------------------------------------------+
void OpenFirstOrder()
{
   MqlTradeRequest request = {};
   MqlTradeResult result = {};
   
   double price;
   ENUM_ORDER_TYPE type;
   
   if(OrderDirection == ORDER_TYPE_BUY)
   {
      price = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      type = ORDER_TYPE_BUY;
   }
   else
   {
      price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      type = ORDER_TYPE_SELL;
   }
   
   request.action = TRADE_ACTION_DEAL;
   request.symbol = _Symbol;
   request.volume = FirstLot;
   request.type = type;
   request.price = price;
   request.deviation = Slippage;
   request.magic = MagicNumber;
   request.comment = TradeComment + "_1";
   
   if(OrderSend(request, result))
   {
      if(result.retcode == TRADE_RETCODE_DONE || result.retcode == TRADE_RETCODE_PLACED)
      {
         Print("First Order Opened: Type=", EnumToString(type), ", Lot=", FirstLot, ", Price=", price);
      }
      else
      {
         Print("Error opening first order: ", result.retcode, " - ", result.comment);
      }
   }
   else
   {
      Print("OrderSend failed: ", GetLastError());
   }
}

//+------------------------------------------------------------------+
//| Check and Open Next Order                                         |
//+------------------------------------------------------------------+
void CheckAndOpenNextOrder()
{
   if(totalOrders == 0) return;
   
   double currentPrice;
   double distance;
   
   if(OrderDirection == ORDER_TYPE_BUY)
   {
      currentPrice = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      distance = lastPrice - currentPrice;
   }
   else
   {
      currentPrice = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      distance = currentPrice - lastPrice;
   }
   
   //--- Convert distance to points
   double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
   int distancePoints = (int)(distance / point);
   
   //--- Check if we need to open next order
   if(distancePoints >= StepPoints)
   {
      double nextLot = CalculateNextLot();
      
      //--- Limit max lot size
      if(nextLot > MaxLotSize)
         nextLot = MaxLotSize;
      
      //--- Normalize lot size
      nextLot = NormalizeLot(nextLot);
      
      if(nextLot > 0)
      {
         OpenNextOrder(nextLot);
      }
   }
}

//+------------------------------------------------------------------+
//| Calculate Next Lot Size using AW Formula                          |
//+------------------------------------------------------------------+
double CalculateNextLot()
{
   double nextLot = 0;
   
   if(UseAWFormula)
   {
      //--- AW Recovery Formula
      //--- Calculate required lot to recover all losses plus target profit
      double currentDrawdown = MathAbs(totalProfit);
      
      if(totalProfit < 0)
      {
         //--- Calculate lot needed to recover
         double pipValue = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
         double pipSize = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
         double stepPips = StepPoints * SymbolInfoDouble(_Symbol, SYMBOL_POINT) / pipSize;
         
         if(stepPips > 0 && pipValue > 0)
         {
            double requiredProfit = currentDrawdown + TakeProfitMoney;
            nextLot = (requiredProfit / (stepPips * pipValue)) * RecoveryFactor;
         }
         else
         {
            nextLot = FirstLot * MathPow(LotMultiplier, totalOrders);
         }
      }
      else
      {
         //--- If in profit, use simple multiplier
         nextLot = FirstLot * MathPow(LotMultiplier, totalOrders);
      }
   }
   else
   {
      //--- Simple multiplier method
      nextLot = FirstLot * MathPow(LotMultiplier, totalOrders);
   }
   
   return nextLot;
}

//+------------------------------------------------------------------+
//| Open Next Order                                                    |
//+------------------------------------------------------------------+
void OpenNextOrder(double lot)
{
   MqlTradeRequest request = {};
   MqlTradeResult result = {};
   
   double price;
   ENUM_ORDER_TYPE type;
   
   if(OrderDirection == ORDER_TYPE_BUY)
   {
      price = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      type = ORDER_TYPE_BUY;
   }
   else
   {
      price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      type = ORDER_TYPE_SELL;
   }
   
   request.action = TRADE_ACTION_DEAL;
   request.symbol = _Symbol;
   request.volume = lot;
   request.type = type;
   request.price = price;
   request.deviation = Slippage;
   request.magic = MagicNumber;
   request.comment = TradeComment + "_" + IntegerToString(totalOrders + 1);
   
   if(OrderSend(request, result))
   {
      if(result.retcode == TRADE_RETCODE_DONE || result.retcode == TRADE_RETCODE_PLACED)
      {
         Print("Order #", totalOrders + 1, " Opened: Lot=", lot, ", Price=", price, 
               ", Total Orders=", totalOrders + 1, ", Current P/L=$", DoubleToString(totalProfit, 2));
      }
      else
      {
         Print("Error opening order: ", result.retcode, " - ", result.comment);
      }
   }
   else
   {
      Print("OrderSend failed: ", GetLastError());
   }
}

//+------------------------------------------------------------------+
//| Close All Orders                                                  |
//+------------------------------------------------------------------+
void CloseAllOrders()
{
   MqlTradeRequest request = {};
   MqlTradeResult result = {};
   
   int closed = 0;
   
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol)
      {
         if(PositionGetInteger(POSITION_MAGIC) == MagicNumber)
         {
            ulong ticket = PositionGetInteger(POSITION_TICKET);
            double volume = PositionGetDouble(POSITION_VOLUME);
            ENUM_POSITION_TYPE posType = (ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE);
            
            request.action = TRADE_ACTION_DEAL;
            request.position = ticket;
            request.symbol = _Symbol;
            request.volume = volume;
            request.deviation = Slippage;
            request.magic = MagicNumber;
            
            if(posType == POSITION_TYPE_BUY)
            {
               request.type = ORDER_TYPE_SELL;
               request.price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
            }
            else
            {
               request.type = ORDER_TYPE_BUY;
               request.price = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
            }
            
            if(OrderSend(request, result))
            {
               if(result.retcode == TRADE_RETCODE_DONE || result.retcode == TRADE_RETCODE_PLACED)
               {
                  closed++;
               }
            }
            
            Sleep(100); // Small delay between closes
         }
      }
   }
   
   Print("=== Closed ", closed, " orders. Final Profit: $", DoubleToString(totalProfit, 2), " ===");
}

//+------------------------------------------------------------------+
//| Normalize Lot Size                                                |
//+------------------------------------------------------------------+
double NormalizeLot(double lot)
{
   double minLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
   double maxLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MAX);
   double lotStep = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
   
   if(lot < minLot) lot = minLot;
   if(lot > maxLot) lot = maxLot;
   
   lot = MathFloor(lot / lotStep) * lotStep;
   
   return lot;
}

//+------------------------------------------------------------------+
```

## คำอธิบาย EA AW Recover

### หลักการทำงาน:

1. **เปิด Order แรก** - ใช้ Lot ตามที่กำหนดใน `FirstLot`

2. **Grid/Averaging System** - เมื่อราคาเคลื่อนไปในทิศทางตรงข้ามตาม `StepPoints` จะเปิด order เพิ่ม

3. **การคำนวณ Lot แบบ AW**:
   - ใช้สูตร: `Lot = (ขาดทุนปัจจุบัน + เป้ากำไร) / (ระยะ Step × มูลค่า Pip) × Recovery Factor`
   - ทำให้เมื่อราคากลับมาตาม Step ที่กำหนด จะทำกำไรได้ทันที

4. **ปิดทั้งหมด** - เมื่อกำไรรวมถึง `TakeProfitMoney`

### พารามิเตอร์สำคัญ:

- **FirstLot** - ขนาด Lot เริ่มต้น
- **LotMultiplier** - ตัวคูณ Lot (ถ้าไม่ใช้สูตร AW)
- **StepPoints** - ระยะห่างในการเปิด order ใหม่
- **MaxOrders** - จำนวน order สูงสุด (จำกัดความเสี่ยง)
- **TakeProfitMoney** - เป้ากำไรเป็นเงิน
- **RecoveryFactor** - ตัวคูณสำหรับการ recover (แนะนำ 1.2-1.5)
- **MaxDrawdownMoney** - ขาดทุนสูงสุดที่ยอมรับได้

### คำเตือน:

⚠️ **ระบบนี้มีความเสี่ยงสูงมาก**
- ใช้เงินทุนที่พร้อมเสีย
- ตั้ง MaxOrders และ MaxDrawdown
- ทดสอบใน Demo Account ก่อน
- Martingale/Grid มีโอกาสเทขาดได้

# EA AW Recover V2 - เพิ่มฟีเจอร์ครบ

```mql5
//+------------------------------------------------------------------+
//|                                         AW_Recover_Advanced.mq5  |
//|                                   Quant Trader - Forex Recovery  |
//+------------------------------------------------------------------+
#property copyright "Quant Trader"
#property version   "2.00"
#property strict

//--- Enums
enum ENUM_ENTRY_MODE
{
   ENTRY_GRID = 0,           // Grid (One Direction)
   ENTRY_FOLLOW_TREND = 1    // Follow Trend (Candle Color)
};

//--- Input Parameters
input group "===== Entry Mode ====="
input ENUM_ENTRY_MODE EntryMode = ENTRY_GRID;           // Entry Mode
input ENUM_ORDER_TYPE GridDirection = ORDER_TYPE_BUY;   // Grid Direction (for Grid mode)

input group "===== Trading Settings ====="
input double FirstLot = 0.01;                           // First Order Lot Size
input double LotMultiplier = 1.5;                       // Lot Multiplier
input int StepPoints = 300;                             // Step Distance (Points)
input int MaxOrders = 10;                               // Maximum Orders
input double TakeProfitMoney = 50;                      // Target Profit ($)

input group "===== Recovery Settings ====="
input bool UseAWFormula = true;                         // Use AW Recovery Formula
input double RecoveryFactor = 1.3;                      // Recovery Factor
input double DrawdownPercentToStart = 5.0;              // Drawdown % to Start Recovery
input bool EnablePartialClose = true;                   // Enable Partial Close
input int PartialCloseAfterOrders = 3;                  // Partial Close After N Orders
input double PartialCloseProfitTarget = 20;             // Partial Close Profit Target ($)

input group "===== Risk Management ====="
input double MaxLotSize = 10.0;                         // Maximum Lot Size
input double MaxDrawdownMoney = 1000;                   // Max Drawdown ($)
input double MaxDrawdownPercent = 30.0;                 // Max Drawdown (%)
input double AccountBalanceForCalc = 0;                 // Account Balance for % Calc (0=Auto)

input group "===== Other Settings ====="
input int MagicNumber = 88888;                          // Magic Number
input int Slippage = 30;                                // Slippage (Points)
input string TradeComment = "AW_Recover";               // Trade Comment

//--- Global Variables
int totalOrders = 0;
double lastPrice = 0;
double averagePrice = 0;
double totalLots = 0;
double totalProfit = 0;
double maxDrawdown = 0;
bool isRecoveryMode = false;
double initialBalance = 0;
datetime lastBarTime = 0;

struct OrderInfo
{
   ulong ticket;
   double lots;
   double openPrice;
   double profit;
   ENUM_POSITION_TYPE type;
};

//+------------------------------------------------------------------+
//| Expert initialization function                                    |
//+------------------------------------------------------------------+
int OnInit()
{
   Print("=== AW Recover Advanced EA V2 Initialized ===");
   Print("Symbol: ", _Symbol);
   Print("Entry Mode: ", EntryMode == ENTRY_GRID ? "Grid" : "Follow Trend");
   Print("First Lot: ", FirstLot);
   Print("Step Points: ", StepPoints);
   Print("Max Orders: ", MaxOrders);
   Print("Target Profit: $", TakeProfitMoney);
   Print("Drawdown % to Start: ", DrawdownPercentToStart, "%");
   Print("Partial Close: ", EnablePartialClose ? "ON" : "OFF");
   
   initialBalance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
   
   return(INIT_SUCCEEDED);
}

//+------------------------------------------------------------------+
//| Expert deinitialization function                                  |
//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
   Print("=== AW Recover Advanced EA Stopped ===");
}

//+------------------------------------------------------------------+
//| Expert tick function                                              |
//+------------------------------------------------------------------+
void OnTick()
{
   //--- Update order info
   UpdateOrderInfo();
   
   //--- Check drawdown percentage
   double accountBalance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
   double currentDrawdownPercent = (totalProfit / accountBalance) * 100;
   
   //--- Update max drawdown
   if(currentDrawdownPercent < maxDrawdown)
      maxDrawdown = currentDrawdownPercent;
   
   //--- Check if we should enter recovery mode
   if(!isRecoveryMode && totalOrders == 0)
   {
      if(MathAbs(currentDrawdownPercent) >= DrawdownPercentToStart)
      {
         isRecoveryMode = true;
         Print("*** RECOVERY MODE ACTIVATED at ", MathAbs(currentDrawdownPercent), "% Drawdown ***");
      }
   }
   
   //--- Check if we need to close all orders (target reached)
   if(totalOrders > 0)
   {
      if(totalProfit >= TakeProfitMoney)
      {
         CloseAllOrders();
         Print("*** TARGET PROFIT REACHED: $", totalProfit, " ***");
         isRecoveryMode = false;
         maxDrawdown = 0;
         return;
      }
      
      //--- Check max drawdown in money
      if(MaxDrawdownMoney > 0 && totalProfit <= -MaxDrawdownMoney)
      {
         CloseAllOrders();
         Print("*** MAX DRAWDOWN (Money) REACHED: $", totalProfit, " ***");
         isRecoveryMode = false;
         maxDrawdown = 0;
         return;
      }
      
      //--- Check max drawdown in percent
      if(MaxDrawdownPercent > 0 && MathAbs(currentDrawdownPercent) >= MaxDrawdownPercent)
      {
         CloseAllOrders();
         Print("*** MAX DRAWDOWN (Percent) REACHED: ", MathAbs(currentDrawdownPercent), "% ***");
         isRecoveryMode = false;
         maxDrawdown = 0;
         return;
      }
      
      //--- Check partial close
      if(EnablePartialClose && totalOrders >= PartialCloseAfterOrders)
      {
         if(totalProfit >= PartialCloseProfitTarget)
         {
            CloseWorstOrder();
         }
      }
   }
   
   //--- Entry logic based on mode
   if(EntryMode == ENTRY_GRID)
   {
      //--- Grid Mode
      if(totalOrders == 0 && isRecoveryMode)
      {
         OpenFirstOrder();
         return;
      }
      
      if(totalOrders > 0 && totalOrders < MaxOrders)
      {
         CheckAndOpenNextOrder();
      }
   }
   else if(EntryMode == ENTRY_FOLLOW_TREND)
   {
      //--- Follow Trend Mode
      if(totalOrders == 0 && isRecoveryMode)
      {
         OpenFirstOrderByTrend();
         return;
      }
      
      if(totalOrders > 0 && totalOrders < MaxOrders)
      {
         CheckAndOpenNextOrderByTrend();
      }
   }
}

//+------------------------------------------------------------------+
//| Update Order Information                                          |
//+------------------------------------------------------------------+
void UpdateOrderInfo()
{
   totalOrders = 0;
   totalLots = 0;
   totalProfit = 0;
   double sumPrice = 0;
   lastPrice = 0;
   
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol)
      {
         if(PositionGetInteger(POSITION_MAGIC) == MagicNumber)
         {
            totalOrders++;
            double lots = PositionGetDouble(POSITION_VOLUME);
            double openPrice = PositionGetDouble(POSITION_PRICE_OPEN);
            double profit = PositionGetDouble(POSITION_PROFIT);
            double swap = PositionGetDouble(POSITION_SWAP);
            double commission = PositionGetDouble(POSITION_COMMISSION);
            
            totalLots += lots;
            sumPrice += openPrice * lots;
            totalProfit += profit + swap + commission;
            
            //--- Keep track of last order price
            if(totalOrders == 1) 
            {
               lastPrice = openPrice;
            }
         }
      }
   }
   
   //--- Calculate average price
   if(totalLots > 0)
      averagePrice = sumPrice / totalLots;
   else
      averagePrice = 0;
}

//| Check and Open Next Order (Grid Mode)                             |
//+------------------------------------------------------------------+
void CheckAndOpenNextOrder()
{
   if(totalOrders == 0) return;
   
   //--- Get last order info
   ENUM_POSITION_TYPE lastType;
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol)
      {
         if(PositionGetInteger(POSITION_MAGIC) == MagicNumber)
         {
            lastPrice = PositionGetDouble(POSITION_PRICE_OPEN);
            lastType = (ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE);
            break;
         }
      }
   }
   
   double currentPrice;
   double distance;
   
   if(lastType == POSITION_TYPE_BUY)
   {
      currentPrice = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      distance = lastPrice - currentPrice;
   }
   else
   {
      currentPrice = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      distance = currentPrice - lastPrice;
   }
   
   //--- Convert distance to points
   double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
   int distancePoints = (int)(distance / point);
   
   //--- Check if we need to open next order
   if(distancePoints >= StepPoints)
   {
      double nextLot = CalculateNextLot();
      
      //--- Limit max lot size
      if(nextLot > MaxLotSize)
         nextLot = MaxLotSize;
      
      //--- Normalize lot size
      nextLot = NormalizeLot(nextLot);
      
      if(nextLot > 0)
      {
         OpenNextOrder(nextLot, lastType == POSITION_TYPE_BUY ? ORDER_TYPE_BUY : ORDER_TYPE_SELL);
      }
   }
}

//| Calculate Next Lot Size using AW Formula                          |
//+------------------------------------------------------------------+
double CalculateNextLot()
{
   double nextLot = 0;
   
   if(UseAWFormula)
   {
      //--- AW Recovery Formula
      double currentDrawdown = MathAbs(totalProfit);
      
      if(totalProfit < 0)
      {
         //--- Calculate lot needed to recover
         double pipValue = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
         double pipSize = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
         double stepPips = StepPoints * SymbolInfoDouble(_Symbol, SYMBOL_POINT) / pipSize;
         
         if(stepPips > 0 && pipValue > 0)
         {
            double requiredProfit = currentDrawdown + TakeProfitMoney;
            nextLot = (requiredProfit / (stepPips * pipValue)) * RecoveryFactor;
         }
         else
         {
            nextLot = FirstLot * MathPow(LotMultiplier, totalOrders);
         }
      }
      else
      {
         //--- If in profit, use simple multiplier
         nextLot = FirstLot * MathPow(LotMultiplier, totalOrders);
      }
   }
   else
   {
      //--- Simple multiplier method
      nextLot = FirstLot * MathPow(LotMultiplier, totalOrders);
   }
   
   return nextLot;
}

//| Normalize Lot Size                                                |
//+------------------------------------------------------------------+
double NormalizeLot(double lot)
{
   double minLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
   double maxLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MAX);
   double lotStep = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
   
   if(lot < minLot) lot = minLot;
   if(lot > maxLot) lot = maxLot;
   
   lot = MathFloor(lot / lotStep) * lotStep;
   
   return lot;
}

   if(atrHandle != INVALID_HANDLE)
      IndicatorRelease(atrHandle);
   
   Print("=== AW Recover Full Featured EA Stopped ===");
}

//+------------------------------------------------------------------+
//| Expert tick function                                              |
//+------------------------------------------------------------------+
void OnTick()
{
   //--- Update order info
   UpdateOrderInfo();
   
   //--- Check drawdown percentage
   double accountBalance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
   double currentDrawdownPercent = (totalProfit / accountBalance) * 100;
   
   //--- Update max drawdown
   if(currentDrawdownPercent < maxDrawdown)
      maxDrawdown = currentDrawdownPercent;
   
   //--- Check if we should enter recovery mode
   if(EnableRecoveryMode && !isRecoveryMode && totalOrders == 0)
   {
      if(MathAbs(currentDrawdownPercent) >= DrawdownPercentToStart)
      {
         isRecoveryMode = true;
         Print("*** RECOVERY MODE ACTIVATED at ", MathAbs(currentDrawdownPercent), "% Drawdown ***");
      }
   }
   
   //--- Check if we need to close all orders (target reached)
   if(totalOrders > 0)
   {
      double targetProfit = CalculateTargetProfit();
      
      if(totalProfit >= targetProfit)
      {
         CloseAllOrders();
         Print("*** TARGET PROFIT REACHED: $", totalProfit, " (Target: $", targetProfit, ") ***");
         isRecoveryMode = false;
         maxDrawdown = 0;
         return;
      }
      
      //--- Check max drawdown in money
      if(MaxDrawdownMoney > 0 && totalProfit <= -MaxDrawdownMoney)
      {
         CloseAllOrders();
         Print("*** MAX DRAWDOWN (Money) REACHED: $", totalProfit, " ***");
         isRecoveryMode = false;
         maxDrawdown = 0;
         return;
      }
      
      //--- Check max drawdown in percent
      if(MaxDrawdownPercent > 0 && MathAbs(currentDrawdownPercent) >= MaxDrawdownPercent)
      {
         CloseAllOrders();
         Print("*** MAX DRAWDOWN (Percent) REACHED: ", MathAbs(currentDrawdownPercent), "% ***");
         isRecoveryMode = false;
         maxDrawdown = 0;
         return;
      }
      
      //--- Check partial close
      if(EnableRecoveryMode && EnablePartialClose && totalOrders >= PartialCloseAfterOrders)
      {
         if(totalProfit >= PartialCloseProfitTarget)
         {
            CloseWorstOrder();
         }
      }
   }
   
   //--- Entry logic based on mode
   if(EntryMode == ENTRY_GRID)
   {
      //--- Grid Mode
      if(totalOrders == 0)
      {
         OpenFirstOrder();
         return;
      }
      
      if(totalOrders > 0 && totalOrders < MaxOrders)
      {
         CheckAndOpenNextOrder();
      }
   }
   else if(EntryMode == ENTRY_FOLLOW_TREND)
   {
      //--- Follow Trend Mode
      if(totalOrders == 0)
      {
         OpenFirstOrderByTrend();
         return;
      }
      
      if(totalOrders > 0 && totalOrders < MaxOrders)
      {
         CheckAndOpenNextOrderByTrend();
      }
   }
   else if(EntryMode == ENTRY_RSI_PA)
   {
      //--- RSI + Price Action Mode
      if(totalOrders == 0 || (!EnableRecoveryMode))
      {
         OpenOrderByRSI_PA();
         return;
      }
      
      if(EnableRecoveryMode && totalOrders > 0 && totalOrders < MaxOrders)
      {
         CheckAndOpenNextOrderByRSI_PA();
      }
   }
}

//+------------------------------------------------------------------+
//| Calculate Target Profit                                           |
//+------------------------------------------------------------------+
double CalculateTargetProfit()
{
   double target = 0;
   
   if(TakeProfitMode == TP_FIXED_PIP)
   {
      //--- Fixed pips
      double pipValue = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
      double pipSize = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
      double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
      
      double pipsInPoints = TakeProfitPips * (pipSize / point);
      target = pipsInPoints * pipValue * totalLots;
   }
   else if(TakeProfitMode == TP_PERCENT_BALANCE)
   {
      //--- Percent of balance
      double balance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
      target = balance * (TakeProfitPercent / 100.0);
   }
   
   return target;
}

//+------------------------------------------------------------------+
//| Calculate Lot Size                                                |
//+------------------------------------------------------------------+
double CalculateLotSize(int orderNumber = 1)
{
   double lot = 0;
   
   if(LotMode == LOT_FIXED)
   {
      //--- Fixed lot with multiplier
      if(EnableRecoveryMode && UseAWFormula && totalOrders > 0)
      {
         lot = CalculateAWLot();
      }
      else
      {
         lot = FirstLot * MathPow(LotMultiplier, orderNumber - 1);
      }
   }
   else if(LotMode == LOT_PERCENT_BALANCE)
   {
      //--- Percent of balance
      double balance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
      lot = (balance / 1000.0) * LotPercentBalance;
      
      if(orderNumber > 1)
      {
         if(EnableRecoveryMode && UseAWFormula && totalOrders > 0)
         {
            lot = CalculateAWLot();
         }
         else
         {
            lot = lot * MathPow(LotMultiplier, orderNumber - 1);
         }
      }
   }
   
   //--- Limit max lot
   if(lot > MaxLotSize)
      lot = MaxLotSize;
   
   return NormalizeLot(lot);
}

//+------------------------------------------------------------------+
//| Calculate AW Recovery Lot                                         |
//+------------------------------------------------------------------+
double CalculateAWLot()
{
   if(totalProfit >= 0)
      return CalculateLotSize(totalOrders + 1);
   
   double currentDrawdown = MathAbs(totalProfit);
   double targetProfit = CalculateTargetProfit();
   double requiredProfit = currentDrawdown + targetProfit;
   
   double stepDistance = GetCurrentStepDistance();
   double pipValue = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
   double pipSize = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
   double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
   
   double stepPips = stepDistance * (point / pipSize);
   
   if(stepPips > 0 && pipValue > 0)
   {
      double lot = (requiredProfit / (stepPips * pipValue)) * RecoveryFactor;
      return lot;
   }
   
   return CalculateLotSize(totalOrders + 1);
}

//+------------------------------------------------------------------+
//| Get Current Step Distance                                         |
//+------------------------------------------------------------------+
double GetCurrentStepDistance()
{
   double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
   
   if(GridType == GRID_FIXED)
   {
      return FixedStepPoints * point;
   }
   else if(GridType == GRID_LINEAR)
   {
      //--- Linear progression
      int step = 0;
      if(totalOrders == 1)
         step = LinearStep1;
      else if(totalOrders == 2)
         step = LinearStep2;
      else if(totalOrders >= 3)
         step = LinearStep3;
      
      return step * point;
   }
   else if(GridType == GRID_DYNAMIC_ATR)
   {
      //--- ATR based
      double atr[];
      ArraySetAsSeries(atr, true);
      
      if(CopyBuffer(atrHandle, 0, 0, 1, atr) > 0)
      {
         return atr[0] * ATR_Multiplier;
      }
      
      return FixedStepPoints * point; // Fallback
   }
   
   return FixedStepPoints * point;
}

//| Update Order Information                                          |
//+------------------------------------------------------------------+
void UpdateOrderInfo()
{
   totalOrders = 0;
   totalLots = 0;
   totalProfit = 0;
   double sumPrice = 0;
   lastPrice = 0;
   
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol)
      {
         if(PositionGetInteger(POSITION_MAGIC) == MagicNumber)
         {
            totalOrders++;
            double lots = PositionGetDouble(POSITION_VOLUME);
            double openPrice = PositionGetDouble(POSITION_PRICE_OPEN);
            double profit = PositionGetDouble(POSITION_PROFIT);
            double swap = PositionGetDouble(POSITION_SWAP);
            double commission = PositionGetDouble(POSITION_COMMISSION);
            
            totalLots += lots;
            sumPrice += openPrice * lots;
            totalProfit += profit + swap + commission;
            
            if(totalOrders == 1) 
            {
               lastPrice = openPrice;
            }
         }
      }
   }
   
   if(totalLots > 0)
      averagePrice = sumPrice / totalLots;
   else
      averagePrice = 0;
}

//+------------------------------------------------------------------+
//| Check RSI and Price Action                                        |
//+------------------------------------------------------------------+
int CheckRSI_PA()
{
   //--- Check new bar
   datetime currentBarTime = iTime(_Symbol, PERIOD_CURRENT, 0);
   if(currentBarTime == lastBarTime) return 0;
   
   //--- Get RSI value
   double rsi[];
   ArraySetAsSeries(rsi, true);
   if(CopyBuffer(rsiHandle, 0, 0, 2, rsi) < 2)
      return 0;
   
   //--- Get candle data
   double open = iOpen(_Symbol, PERIOD_CURRENT, 1);
   double close = iClose(_Symbol, PERIOD_CURRENT, 1);
   double high = iHigh(_Symbol, PERIOD_CURRENT, 1);
   double low = iLow(_Symbol, PERIOD_CURRENT, 1);
   
   double candleSize = high - low;
   double bodySize = MathAbs(close - open);
   double bodyPercent = (bodySize / candleSize) * 100;
   
   //--- Check for BUY signal
   if(rsi[1] < RSI_Oversold)
   {
      if(RequireBullishCandle)
      {
         if(close > open && bodyPercent >= MinCandleBodyPercent)
         {
            return 1; // BUY signal
         }
      }
      else
      {
         return 1; // BUY signal
      }
   }
   
   //--- Check for SELL signal
   if(rsi[1] > RSI_Overbought)
   {
      if(RequireBearishCandle)
      {
         if(close < open && bodyPercent >= MinCandleBodyPercent)
         {
            return -1; // SELL signal
         }
      }
      else
      {
         return -1; // SELL signal
      }
   }
   
   return 0; // No signal
}

void CloseWorstOrder()
{
   double worstProfit = 0;
   ulong worstTicket = 0;
   double worstVolume = 0;
   ENUM_POSITION_TYPE worstType;
   
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol)
      {
         if(PositionGetInteger(POSITION_MAGIC) == MagicNumber)
         {
            double profit = PositionGetDouble(POSITION_PROFIT);
            
            if(profit < worstProfit || worstTicket == 0)
            {
               worstProfit = profit;
               worstTicket = PositionGetInteger(POSITION_TICKET);
               worstVolume = PositionGetDouble(POSITION_VOLUME);
               worstType = (ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE);
            }
         }
      }
   }
   
   if(worstTicket > 0)
   {
      MqlTradeRequest request = {};
      MqlTradeResult result = {};
      
      request.action = TRADE_ACTION_DEAL;
      request.position = worstTicket;
      request.symbol = _Symbol;
      request.volume = worstVolume;
      request.deviation = Slippage;
      request.magic = MagicNumber;
      
      if(worstType == POSITION_TYPE_BUY)
      {
         request.type = ORDER_TYPE_SELL;
         request.price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      }
      else
      {
         request.type = ORDER_TYPE_BUY;
         request.price = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      }
      
      if(OrderSend(request, result))
      {
         if(result.retcode == TRADE_RETCODE_DONE || result.retcode == TRADE_RETCODE_PLACED)
         {
            Print("*** PARTIAL CLOSE: Closed worst order #", worstTicket, " with loss $", worstProfit, " ***");
         }
      }
   }
}

//| Normalize Lot Size                                                |
//+------------------------------------------------------------------+
double NormalizeLot(double lot)
{
   double minLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
   double maxLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MAX);
   double lotStep = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
   
   if(lot < minLot) lot = minLot;
   if(lot > maxLot) lot = maxLot;
   
   lot = MathFloor(lot / lotStep) * lotStep;
   
   return lot;
}

//+------------------------------------------------------------------+
//| Expert tick function                                              |
//+------------------------------------------------------------------+
void OnTick()
{
   //--- Update order info
   UpdateOrderInfo();
   
   //--- Check for reset condition
   if(AutoResetOnWrong && needsReset)
   {
      if(totalProfit <= ResetLossThreshold)
      {
         Print("*** AUTO RESET: Closing all positions. Loss: $", totalProfit, " ***");
         CloseAllOrders();
         needsReset = false;
         isRecoveryMode = false;
         return;
      }
   }
   
   //--- Check drawdown percentage
   double accountBalance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
   double currentDrawdownPercent = (totalProfit / accountBalance) * 100;
   
   if(currentDrawdownPercent < maxDrawdown)
      maxDrawdown = currentDrawdownPercent;
   
   //--- Check if we should enter recovery mode
   if(EnableRecoveryMode && !isRecoveryMode && totalOrders == 0)
   {
      if(MathAbs(currentDrawdownPercent) >= DrawdownPercentToStart)
      {
         isRecoveryMode = true;
         Print("*** RECOVERY MODE ACTIVATED at ", MathAbs(currentDrawdownPercent), "% Drawdown ***");
      }
   }
   
   //--- Check if we need to close all orders (target reached)
   if(totalOrders > 0)
   {
      double targetProfit = CalculateTargetProfit();
      
      if(totalProfit >= targetProfit)
      {
         CloseAllOrders();
         Print("*** TARGET PROFIT REACHED: $", totalProfit, " (Target: $", targetProfit, ") ***");
         isRecoveryMode = false;
         maxDrawdown = 0;
         needsReset = false;
         return;
      }
      
      //--- Check max drawdown
      if(MaxDrawdownMoney > 0 && totalProfit <= -MaxDrawdownMoney)
      {
         CloseAllOrders();
         Print("*** MAX DRAWDOWN (Money) REACHED: $", totalProfit, " ***");
         isRecoveryMode = false;
         maxDrawdown = 0;
         needsReset = false;
         return;
      }
      
      if(MaxDrawdownPercent > 0 && MathAbs(currentDrawdownPercent) >= MaxDrawdownPercent)
      {
         CloseAllOrders();
         Print("*** MAX DRAWDOWN (Percent) REACHED: ", MathAbs(currentDrawdownPercent), "% ***");
         isRecoveryMode = false;
         maxDrawdown = 0;
         needsReset = false;
         return;
      }
      
      //--- Check partial close
      if(EnableRecoveryMode && EnablePartialClose && totalOrders >= PartialCloseAfterOrders)
      {
         if(totalProfit >= PartialCloseProfitTarget)
         {
            CloseWorstOrder();
         }
      }
   }
   
   //--- Entry logic based on mode
   if(EntryMode == ENTRY_GRID)
   {
      if(totalOrders == 0)
      {
         OpenFirstOrder();
         return;
      }
      
      if(totalOrders > 0 && totalOrders < MaxOrders)
      {
         CheckAndOpenNextOrder();
      }
   }
   else if(EntryMode == ENTRY_FOLLOW_TREND)
   {
      if(totalOrders == 0)
      {
         OpenFirstOrderByTrend();
         return;
      }
      
      if(totalOrders > 0 && totalOrders < MaxOrders)
      {
         CheckAndOpenNextOrderByTrend();
      }
   }
   else if(EntryMode == ENTRY_RSI_PA)
   {
      if(totalOrders == 0 || (!EnableRecoveryMode))
      {
         OpenOrderByRSI_PA();
         return;
      }
      
      if(EnableRecoveryMode && totalOrders > 0 && totalOrders < MaxOrders)
      {
         CheckAndOpenNextOrderByRSI_PA();
      }
   }
}

//| Calculate Target Profit                                           |
//+------------------------------------------------------------------+
double CalculateTargetProfit()
{
   double target = 0;
   
   if(TakeProfitMode == TP_FIXED_PIP)
   {
      double pipValue = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
      double pipSize = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
      double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
      
      double pipsInPoints = TakeProfitPips * (pipSize / point);
      target = pipsInPoints * pipValue * totalLots;
   }
   else if(TakeProfitMode == TP_PERCENT_BALANCE)
   {
      double balance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
      target = balance * (TakeProfitPercent / 100.0);
   }
   
   return target;
}

//+------------------------------------------------------------------+
//| Calculate Lot Size (Main Entry)                                   |
//+------------------------------------------------------------------+
double CalculateLotSize(int orderNumber = 1)
{
   double lot = 0;
   
   if(LotMode == LOT_FIXED)
   {
      lot = FirstLot * MathPow(LotMultiplier, orderNumber - 1);
   }
   else if(LotMode == LOT_PERCENT_BALANCE)
   {
      double balance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
      lot = (balance / 1000.0) * LotPercentBalance;
      
      if(orderNumber > 1)
         lot = lot * MathPow(LotMultiplier, orderNumber - 1);
   }
   
   if(lot > MaxLotSize)
      lot = MaxLotSize;
   
   return NormalizeLot(lot);
}

//+------------------------------------------------------------------+
//| Calculate Recovery Lot (Advanced Methods)                         |
//+------------------------------------------------------------------+
double CalculateRecoveryLot(ENUM_ORDER_TYPE desiredType)
{
   if(!EnableRecoveryMode || totalOrders == 0)
      return CalculateLotSize(totalOrders + 1);
   
   double lot = 0;
   double balance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
   double currentDrawdown = MathAbs(totalProfit);
   double currentDrawdownPercent = (totalProfit / balance) * 100;
   
   switch(RecoveryLotMode)
   {
      case RECOVERY_AW_FORMULA:
         lot = CalculateAWLot();
         break;
         
      case RECOVERY_DRAWDOWN_BASED:
         // $100 loss = 0.01 lot (default)
         lot = (currentDrawdown / DrawdownPerLot) * 0.01;
         lot = MathMax(lot, FirstLot);
         Print("Recovery Lot (Drawdown Based): $", currentDrawdown, " → ", lot, " lot");
         break;
         
      case RECOVERY_PERCENT_DRAWDOWN:
         // 1% drawdown = 0.01 lot (default)
         lot = (MathAbs(currentDrawdownPercent) / DrawdownPercentPerLot) * 0.01;
         lot = MathMax(lot, FirstLot);
         Print("Recovery Lot (Percent Based): ", MathAbs(currentDrawdownPercent), "% → ", lot, " lot");
         break;
         
      case RECOVERY_OPPOSITE_PERCENT:
         // Opposite direction hedge
         if(EnableOppositeHedge)
         {
            // Calculate opposite lot
            if(desiredType == ORDER_TYPE_BUY && totalSellLots > totalBuyLots)
            {
               // Want to BUY, but more SELL lots → hedge
               lot = totalSellLots * (OppositePercentOfTotal / 100.0);
               Print("Hedge BUY: Total SELL=", totalSellLots, " → BUY ", lot, " lot (", OppositePercentOfTotal, "%)");
            }
            else if(desiredType == ORDER_TYPE_SELL && totalBuyLots > totalSellLots)
            {
               // Want to SELL, but more BUY lots → hedge
               lot = totalBuyLots * (OppositePercentOfTotal / 100.0);
               Print("Hedge SELL: Total BUY=", totalBuyLots, " → SELL ", lot, " lot (", OppositePercentOfTotal, "%)");
            }
            else
            {
               // Same direction → use martingale
               lot = CalculateLotSize(totalOrders + 1);
            }
            
            // Mark for potential reset
            if(totalProfit < 0)
               needsReset = true;
         }
         else
         {
            lot = CalculateLotSize(totalOrders + 1);
         }
         break;
         
      case RECOVERY_MARTINGALE:
         lot = CalculateLotSize(totalOrders + 1);
         break;
   }
   
   if(lot > MaxLotSize)
      lot = MaxLotSize;
   
   return NormalizeLot(lot);
}

//+------------------------------------------------------------------+
//| Calculate AW Recovery Lot                                         |
//+------------------------------------------------------------------+
double CalculateAWLot()
{
   if(totalProfit >= 0)
      return CalculateLotSize(totalOrders + 1);
   
   double currentDrawdown = MathAbs(totalProfit);
   double targetProfit = CalculateTargetProfit();
   double requiredProfit = currentDrawdown + targetProfit;
   
   double stepDistance = GetCurrentStepDistance();
   double pipValue = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
   double pipSize = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
   double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
   
   double stepPips = stepDistance * (point / pipSize);
   
   if(stepPips > 0 && pipValue > 0)
   {
      double lot = (requiredProfit / (stepPips * pipValue)) * RecoveryFactor;
      return lot;
   }
   
   return CalculateLotSize(totalOrders + 1);
}

//+------------------------------------------------------------------+
//| Get Current Step Distance                                         |
//+------------------------------------------------------------------+
double GetCurrentStepDistance()
{
   double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
   
   if(GridType == GRID_FIXED)
   {
      return FixedStepPoints * point;
   }
   else if(GridType == GRID_LINEAR)
   {
      int step = 0;
      if(totalOrders == 1)
         step = LinearStep1;
      else if(totalOrders == 2)
         step = LinearStep2;
      else if(totalOrders >= 3)
         step = LinearStep3;
      
      return step * point;
   }
   else if(GridType == GRID_DYNAMIC_ATR)
   {
      double atr[];
      ArraySetAsSeries(atr, true);
      
      if(CopyBuffer(atrHandle, 0, 0, 1, atr) > 0)
      {
         return atr[0] * ATR_Multiplier;
      }
      
      return FixedStepPoints * point;
   }
   
   return FixedStepPoints * point;
}

//+------------------------------------------------------------------+
//| Update Order Information                                          |
//+------------------------------------------------------------------+
void UpdateOrderInfo()
{
   totalOrders = 0;
   totalBuyOrders = 0;
   totalSellOrders = 0;
   totalBuyLots = 0;
   totalSellLots = 0;
   totalLots = 0;
   totalProfit = 0;
   double sumPrice = 0;
   lastPrice = 0;
   
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol)
      {
         if(PositionGetInteger(POSITION_MAGIC) == MagicNumber)
         {
            totalOrders++;
            double lots = PositionGetDouble(POSITION_VOLUME);
            double openPrice = PositionGetDouble(POSITION_PRICE_OPEN);
            double profit = PositionGetDouble(POSITION_PROFIT);
            double swap = PositionGetDouble(POSITION_SWAP);
            double commission = PositionGetDouble(POSITION_COMMISSION);
            ENUM_POSITION_TYPE posType = (ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE);
            
            totalLots += lots;
            sumPrice += openPrice * lots;
            totalProfit += profit + swap + commission;
            
            if(posType == POSITION_TYPE_BUY)
            {
               totalBuyOrders++;
               totalBuyLots += lots;
            }
            else
            {
               totalSellOrders++;
               totalSellLots += lots;
            }
            
            if(totalOrders == 1) 
            {
               lastPrice = openPrice;
            }
         }
      }
   }
   
   if(totalLots > 0)
      averagePrice = sumPrice / totalLots;
   else
      averagePrice = 0;
}

//+------------------------------------------------------------------+
//| Check RSI and Price Action                                        |
//+------------------------------------------------------------------+
int CheckRSI_PA()
{
   datetime currentBarTime = iTime(_Symbol, PERIOD_CURRENT, 0);
   if(currentBarTime == lastBarTime) return 0;
   
   double rsi[];
   ArraySetAsSeries(rsi, true);
   if(CopyBuffer(rsiHandle, 0, 0, 2, rsi) < 2)
      return 0;
   
   double open = iOpen(_Symbol, PERIOD_CURRENT, 1);
   double close = iClose(_Symbol, PERIOD_CURRENT, 1);
   double high = iHigh(_Symbol, PERIOD_CURRENT, 1);
   double low = iLow(_Symbol, PERIOD_CURRENT, 1);
   
   double candleSize = high - low;
   double bodySize = MathAbs(close - open);
   double bodyPercent = (bodySize / candleSize) * 100;
   
   if(rsi[1] < RSI_Oversold)
   {
      if(RequireBullishCandle)
      {
         if(close > open && bodyPercent >= MinCandleBodyPercent)
            return 1;
      }
      else
      {
         return 1;
      }
   }
   
   if(rsi[1] > RSI_Overbought)
   {
      if(RequireBearishCandle)
      {
         if(close < open && bodyPercent >= MinCandleBodyPercent)
            return -1;
      }
      else
      {
         return -1;
      }
   }
   
   return 0;
}

void CloseWorstOrder()
{
   double worstProfit = 0;
   ulong worstTicket = 0;
   double worstVolume = 0;
   ENUM_POSITION_TYPE worstType;
   
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol)
      {
         if(PositionGetInteger(POSITION_MAGIC) == MagicNumber)
         {
            double profit = PositionGetDouble(POSITION_PROFIT);
            
            if(profit < worstProfit || worstTicket == 0)
            {
               worstProfit = profit;
               worstTicket = PositionGetInteger(POSITION_TICKET);
               worstVolume = PositionGetDouble(POSITION_VOLUME);
               worstType = (ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE);
            }
         }
      }
   }
   
   if(worstTicket > 0)
   {
      MqlTradeRequest request = {};
      MqlTradeResult result = {};
      
      request.action = TRADE_ACTION_DEAL;
      request.position = worstTicket;
      request.symbol = _Symbol;
      request.volume = worstVolume;
      request.deviation = Slippage;
      request.magic = MagicNumber;
      
      if(worstType == POSITION_TYPE_BUY)
      {
         request.type = ORDER_TYPE_SELL;
         request.price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      }
      else
      {
         request.type = ORDER_TYPE_BUY;
         request.price = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      }
      
      if(OrderSend(request, result))
      {
         if(result.retcode == TRADE_RETCODE_DONE || result.retcode == TRADE_RETCODE_PLACED)
         {
            Print("*** PARTIAL CLOSE: Closed worst order #", worstTicket, " with loss $", worstProfit, " ***");
         }
      }
   }
}

//| Normalize Lot Size                                                |
//+------------------------------------------------------------------+
double NormalizeLot(double lot)
{
   double minLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
   double maxLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MAX);
   double lotStep = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
   
   if(lot < minLot) lot = minLot;
   if(lot > maxLot) lot = maxLot;
   
   lot = MathFloor(lot / lotStep) * lotStep;
   
   return lot;
}

## 📊 คำอธิบายแต่ละ Recovery Lot Mode

### 1. ✅ **AW Formula (Original)** - แนะนำสำหรับ Grid
```
RecoveryLotMode = RECOVERY_AW_FORMULA
RecoveryFactor = 1.3
```
**หลักการ:**
- คำนวณ Lot ที่ต้องการเพื่อ recover ขาดทุน + กำไรเป้าหมาย
- ใช้สูตร: `Lot = (Loss + Target) / (Step × PipValue) × Factor`

**✅ ข้อดี:**
- Recover เร็วเมื่อราคากลับตัว
- มีเป้าหมายชัดเจน

**⚠️ ข้อเสีย:**
- Lot เพิ่มเร็วมากใน deep drawdown
- ความเสี่ยงสูงถ้า Factor มาก

**เหมาะกับ:** Grid แนวรับ-แนวต้าน

---

### 2. 💵 **Drawdown Based ($)**
```
RecoveryLotMode = RECOVERY_DRAWDOWN_BASED
DrawdownPerLot = 100  // $100 loss = +0.01 lot
```
**หลักการ:**
- ขาดทุน $100 → เพิ่ม 0.01 lot
- ขาดทุน $500 → ใช้ 0.05 lot

**✅ ข้อดี:**
- เข้าใจง่าย คำนวณตรงไปตรงมา
- ปรับตามขาดทุนจริง

**⚠️ ข้อเสีย:**
- อาจเพิ่ม Lot เร็วเกินไปใน trending market
- ไม่คำนึงถึงขนาด account

**เหมาะกับ:** Account ใหญ่, ranging market

---

### 3. 📈 **Percent Drawdown Based**
```
RecoveryLotMode = RECOVERY_PERCENT_DRAWDOWN
DrawdownPercentPerLot = 1.0  // 1% drawdown = +0.01 lot
```
**หลักการ:**
- Drawdown 5% → ใช้ 0.05 lot
- Drawdown 10% → ใช้ 0.10 lot

**✅ ข้อดี:**
- ปรับตามขนาด account
- เหมาะกับการจัดการเงินแบบ %

**⚠️ ข้อเสีย:**
- อาจรุนแรงเกินไปถ้า % สูง
- ต้องระวังเรื่อง compound

**เหมาะกับ:** Account ทุกขนาด, ต้องการความเสี่ยงเป็น %

### 4. 🔄 **Opposite Direction Hedge** - แนะนำ! 🌟
```
RecoveryLotMode = RECOVERY_OPPOSITE_PERCENT
OppositePercentOfTotal = 30  // 30% ของ lot ที่ค้าง
EnableOppositeHedge = true
AutoResetOnWrong = true
ResetLossThreshold = -50
```
**หลักการ:**
- ค้าง BUY 1.0 lot → เปิด SELL 0.3 lot
- ค้าง SELL 0.5 lot → เปิด BUY 0.15 lot
- ถ้าขาดทุนเกิน $50 → ปิดทั้งหมด reset

**✅ ข้อดี:**
- 🛡️ **Hedging ลดความเสี่ยง**
- 💰 กำไรได้ทั้ง 2 ทิศทาง
- 🔄 Auto reset ป้องกัน blow up
- เหมาะกับตลาดไม่แน่นอน

**⚠️ ข้อเสีย:**
- อาจพลาดกำไรเต็มจาก trend
- ต้องมี netting account (MT5)
- Spread/Commission เยอะขึ้น

**เหมาะกับ:** Choppy market, ไม่แน่ใจทิศทาง

**ตัวอย่าง:**
```
Scenario 1: Trend ผิด
- เปิด BUY 1.0 lot ที่ 1.1000
- ราคาลง 1.0950 → เปิด SELL 0.3 lot (hedge)
- ราคาลงต่อ 1.0900 → SELL profit, BUY loss
- Net loss ลดลง!

Scenario 2: Trend ถูก
- เปิด BUY 1.0 lot ที่ 1.1000
- ราคาลง 1.0950 → เปิด SELL 0.3 lot
- ราคากลับขึ้น 1.1050 → BUY profit มาก, SELL loss เล็กน้อย
- Net profit ดี!
```

   RECOVERY_AW_FORMULA = 0,          // AW Formula (Original)
   RECOVERY_DRAWDOWN_BASED = 1,      // Based on Drawdown ($)
   RECOVERY_PERCENT_DRAWDOWN = 2,    // Based on Drawdown (%)
   RECOVERY_OPPOSITE_PERCENT = 3,    // Opposite Direction % of Total Lots
   RECOVERY_MARTINGALE = 4           // Simple Martingale
};

input double MaxDrawdownMoney = 1000;                  // Max Drawdown ($)

//+------------------------------------------------------------------+
//| Expert initialization function                                    |
//+------------------------------------------------------------------+
int OnInit()
{
   Print("=== AW Recover Improved Grid System V4.1 Initialized ===");
   Print("Symbol: ", _Symbol);
   Print("Entry Mode: ", EnumToString(EntryMode));
   Print("Grid Type: ", EnumToString(GridType));
   PrintGridConfiguration();
   Print("Lot Mode: ", EnumToString(LotMode));
   Print("Recovery Lot Mode: ", EnumToString(RecoveryLotMode));
   Print("TP Mode: ", EnumToString(TakeProfitMode));
   Print("Recovery Mode: ", EnableRecoveryMode ? "ON" : "OFF");
   
   //--- Initialize indicators
   rsiHandle = iRSI(_Symbol, PERIOD_CURRENT, RSI_Period, PRICE_CLOSE);
   if(rsiHandle == INVALID_HANDLE)
   {
      Print("Error creating RSI indicator");
      return(INIT_FAILED);
   }
   
   atrHandle = iATR(_Symbol, ATR_Timeframe, ATR_Period);
   if(atrHandle == INVALID_HANDLE)
   {
      Print("Error creating ATR indicator");
      return(INIT_FAILED);
   }
   
   initialBalance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
   
   return(INIT_SUCCEEDED);
}

//+------------------------------------------------------------------+
//| Print Grid Configuration                                          |
//+------------------------------------------------------------------+
void PrintGridConfiguration()
{
   Print("=== Grid Configuration ===");
   
   if(GridType == GRID_FIXED)
   {
      Print("Mode: FIXED");
      Print("- Distance: ", FixedStepPoints, " points (constant)");
   }
   else if(GridType == GRID_LINEAR)
   {
      Print("Mode: LINEAR (Distance × Order Number)");
      Print("- Base Step: ", LinearBaseStep, " points");
      Print("- Multiply by Order: ", LinearMultiplyByOrder ? "YES" : "NO");
      Print("- Growth Factor: ", LinearGrowthFactor);
      
      if(LinearMultiplyByOrder)
      {
         Print("📊 Example Grid Distances:");
         for(int i = 1; i <= 5; i++)
         {
            double dist = CalculateLinearDistance(i);
            Print("  Order ", i, ": ", (int)(dist / SymbolInfoDouble(_Symbol, SYMBOL_POINT)), " points");
         }
      }
   }
   else if(GridType == GRID_DYNAMIC_ATR)
   {
      Print("Mode: DYNAMIC ATR (ATR × Order Number)");

      Print("- ATR Timeframe: ", EnumToString(ATR_Timeframe));
      Print("- ATR Period: ", ATR_Period);

      Print("- Growth Factor: ", ATR_GrowthFactor);

         if(ATR_MultiplyByOrder)
         {
            Print("📊 Example Grid Distances (with current ATR):");
            for(int i = 1; i <= 5; i++)
            {
               double dist = CalculateDynamicATRDistance(i, atr[0]);
               Print("  Order ", i, ": ", (int)(dist / SymbolInfoDouble(_Symbol, SYMBOL_POINT)), " points");
            }
         }
      }
   }
   
   Print("========================");
}

//+------------------------------------------------------------------+
//| Calculate Linear Distance                                         |
//+------------------------------------------------------------------+
double CalculateLinearDistance(int orderNumber)
{
   double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
   
   if(LinearMultiplyByOrder)
   {
      // Distance = BaseStep × (OrderNumber ^ GrowthFactor)
      double multiplier = MathPow(orderNumber, LinearGrowthFactor);
      return LinearBaseStep * multiplier * point;
   }
   else
   {
      // Fixed linear step
      return LinearBaseStep * point;
   }
}

//+------------------------------------------------------------------+
//| Calculate Dynamic ATR Distance                                    |
//+------------------------------------------------------------------+
double CalculateDynamicATRDistance(int orderNumber, double atrValue)
{
   if(ATR_MultiplyByOrder)
   {
      // Distance = ATR × Multiplier × (OrderNumber ^ GrowthFactor)
      double multiplier = MathPow(orderNumber, ATR_GrowthFactor);
      return atrValue * ATR_Multiplier * multiplier;
   }
   else
   {
      // Fixed ATR-based distance
      return atrValue * ATR_Multiplier;
   }
}

//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
   if(rsiHandle != INVALID_HANDLE)
      IndicatorRelease(rsiHandle);
   if(atrHandle != INVALID_HANDLE)
      IndicatorRelease(atrHandle);
   
   Print("=== AW Recover Improved Grid System Stopped ===");
}

//+------------------------------------------------------------------+
//| Expert tick function                                              |
//+------------------------------------------------------------------+
void OnTick()
{
   //--- Update order info
   UpdateOrderInfo();
   
   //--- Check for reset condition
   if(AutoResetOnWrong && needsReset)
   {
      if(totalProfit <= ResetLossThreshold)
      {
         Print("*** AUTO RESET: Closing all positions. Loss: $", totalProfit, " ***");
         CloseAllOrders();
         needsReset = false;
         isRecoveryMode = false;
         return;
      }
   }
   
   //--- Check drawdown percentage
   double accountBalance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
   double currentDrawdownPercent = (totalProfit / accountBalance) * 100;
   
   if(currentDrawdownPercent < maxDrawdown)
      maxDrawdown = currentDrawdownPercent;
   
   //--- Check if we should enter recovery mode
   if(EnableRecoveryMode && !isRecoveryMode && totalOrders == 0)
   {
      if(MathAbs(currentDrawdownPercent) >= DrawdownPercentToStart)
      {
         isRecoveryMode = true;
         Print("*** RECOVERY MODE ACTIVATED at ", MathAbs(currentDrawdownPercent), "% Drawdown ***");
      }
   }
   
   //--- Check if we need to close all orders (target reached)
   if(totalOrders > 0)
   {
      double targetProfit = CalculateTargetProfit();
      
      if(totalProfit >= targetProfit)
      {
         CloseAllOrders();
         Print("*** TARGET PROFIT REACHED: $", totalProfit, " (Target: $", targetProfit, ") ***");
         isRecoveryMode = false;
         maxDrawdown = 0;
         needsReset = false;
         return;
      }
      
      //--- Check max drawdown
      if(MaxDrawdownMoney > 0 && totalProfit <= -MaxDrawdownMoney)
      {
         CloseAllOrders();
         Print("*** MAX DRAWDOWN (Money) REACHED: $", totalProfit, " ***");
         isRecoveryMode = false;
         maxDrawdown = 0;
         needsReset = false;
         return;
      }
      
      if(MaxDrawdownPercent > 0 && MathAbs(currentDrawdownPercent) >= MaxDrawdownPercent)
      {
         CloseAllOrders();
         Print("*** MAX DRAWDOWN (Percent) REACHED: ", MathAbs(currentDrawdownPercent), "% ***");
         isRecoveryMode = false;
         maxDrawdown = 0;
         needsReset = false;
         return;
      }
      
      //--- Check partial close
      if(EnableRecoveryMode && EnablePartialClose && totalOrders >= PartialCloseAfterOrders)
      {
         if(totalProfit >= PartialCloseProfitTarget)
         {
            CloseWorstOrder();
         }
      }
   }
   
   //--- Entry logic based on mode
   if(EntryMode == ENTRY_GRID)
   {
      if(totalOrders == 0)
      {
         OpenFirstOrder();
         return;
      }
      
      if(totalOrders > 0 && totalOrders < MaxOrders)
      {
         CheckAndOpenNextOrder();
      }
   }
   else if(EntryMode == ENTRY_FOLLOW_TREND)
   {
      if(totalOrders == 0)
      {
         OpenFirstOrderByTrend();
         return;
      }
      
      if(totalOrders > 0 && totalOrders < MaxOrders)
      {
         CheckAndOpenNextOrderByTrend();
      }
   }
   else if(EntryMode == ENTRY_RSI_PA)
   {
      if(totalOrders == 0 || (!EnableRecoveryMode))
      {
         OpenOrderByRSI_PA();
         return;
      }
      
      if(EnableRecoveryMode && totalOrders > 0 && totalOrders < MaxOrders)
      {
         CheckAndOpenNextOrderByRSI_PA();
      }
   }
}

//+------------------------------------------------------------------+
//| Get Current Step Distance (Improved)                              |
//+------------------------------------------------------------------+
double GetCurrentStepDistance()
{
   double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
   int nextOrderNumber = totalOrders + 1; // For next order
   
   if(GridType == GRID_FIXED)
   {
      return FixedStepPoints * point;
   }
   else if(GridType == GRID_LINEAR)
   {
      double distance = CalculateLinearDistance(nextOrderNumber);
      
      if(ShowGridInfo)
      {
         int distancePoints = (int)(distance / point);
         Print("📏 Next Grid Distance (Order #", nextOrderNumber, "): ", distancePoints, " points");
      }
      
      return distance;
   }
   else if(GridType == GRID_DYNAMIC_ATR)
   {
      double atr[];
      ArraySetAsSeries(atr, true);
      
      if(CopyBuffer(atrHandle, 0, 0, 1, atr) > 0)
      {
         double distance = CalculateDynamicATRDistance(nextOrderNumber, atr[0]);
         
         if(ShowGridInfo)
         {
            int distancePoints = (int)(distance / point);
            Print("📏 Next Grid Distance (Order #", nextOrderNumber, ", ATR=", atr[0], "): ", distancePoints, " points");
         }
         
         return distance;
      }
      
      return FixedStepPoints * point; // Fallback
   }
   
   return FixedStepPoints * point;
}

//+------------------------------------------------------------------+
//| Calculate Target Profit                                           |
//+------------------------------------------------------------------+
double CalculateTargetProfit()
{
   double target = 0;
   
   if(TakeProfitMode == TP_FIXED_PIP)
   {
      double pipValue = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
      double pipSize = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
      double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
      
      double pipsInPoints = TakeProfitPips * (pipSize / point);
      target = pipsInPoints * pipValue * totalLots;
   }
   else if(TakeProfitMode == TP_PERCENT_BALANCE)
   {
      double balance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
      target = balance * (TakeProfitPercent / 100.0);
   }
   
   return target;
}

//+------------------------------------------------------------------+
//| Calculate Lot Size (Main Entry)                                   |
//+------------------------------------------------------------------+
double CalculateLotSize(int orderNumber = 1)
{
   double lot = 0;
   
   if(LotMode == LOT_FIXED)
   {
      lot = FirstLot * MathPow(LotMultiplier, orderNumber - 1);
   }
   else if(LotMode == LOT_PERCENT_BALANCE)
   {
      double balance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
      lot = (balance / 1000.0) * LotPercentBalance;
      
      if(orderNumber > 1)
         lot = lot * MathPow(LotMultiplier, orderNumber - 1);
   }
   
   if(lot > MaxLotSize)
      lot = MaxLotSize;
   
   return NormalizeLot(lot);
}

//+------------------------------------------------------------------+
//| Calculate Recovery Lot (Advanced Methods)                         |
//+------------------------------------------------------------------+
double CalculateRecoveryLot(ENUM_ORDER_TYPE desiredType)
{
   if(!EnableRecoveryMode || totalOrders == 0)
      return CalculateLotSize(totalOrders + 1);
   
   double lot = 0;
   double balance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
   double currentDrawdown = MathAbs(totalProfit);
   double currentDrawdownPercent = (totalProfit / balance) * 100;
   
   switch(RecoveryLotMode)
   {
      case RECOVERY_AW_FORMULA:
         lot = CalculateAWLot();
         break;
         
      case RECOVERY_DRAWDOWN_BASED:
         lot = (currentDrawdown / DrawdownPerLot) * 0.01;
         lot = MathMax(lot, FirstLot);
         Print("Recovery Lot (Drawdown Based): $", currentDrawdown, " → ", lot, " lot");
         break;
         
      case RECOVERY_PERCENT_DRAWDOWN:
         lot = (MathAbs(currentDrawdownPercent) / DrawdownPercentPerLot) * 0.01;
         lot = MathMax(lot, FirstLot);
         Print("Recovery Lot (Percent Based): ", MathAbs(currentDrawdownPercent), "% → ", lot, " lot");
         break;
         
      case RECOVERY_OPPOSITE_PERCENT:
         if(EnableOppositeHedge)
         {
            if(desiredType == ORDER_TYPE_BUY && totalSellLots > totalBuyLots)
            {
               lot = totalSellLots * (OppositePercentOfTotal / 100.0);
               Print("Hedge BUY: Total SELL=", totalSellLots, " → BUY ", lot, " lot (", OppositePercentOfTotal, "%)");
            }
            else if(desiredType == ORDER_TYPE_SELL && totalBuyLots > totalSellLots)
            {
               lot = totalBuyLots * (OppositePercentOfTotal / 100.0);
               Print("Hedge SELL: Total BUY=", totalBuyLots, " → SELL ", lot, " lot (", OppositePercentOfTotal, "%)");
            }
            else
            {
               lot = CalculateLotSize(totalOrders + 1);
            }
            
            if(totalProfit < 0)
               needsReset = true;
         }
         else
         {
            lot = CalculateLotSize(totalOrders + 1);
         }
         break;
         
      case RECOVERY_MARTINGALE:
         lot = CalculateLotSize(totalOrders + 1);
         break;
   }
   
   if(lot > MaxLotSize)
      lot = MaxLotSize;
   
   return NormalizeLot(lot);
}

//+------------------------------------------------------------------+
//| Calculate AW Recovery Lot                                         |
//+------------------------------------------------------------------+
double CalculateAWLot()
{
   if(totalProfit >= 0)
      return CalculateLotSize(totalOrders + 1);
   
   double currentDrawdown = MathAbs(totalProfit);
   double targetProfit = CalculateTargetProfit();
   double requiredProfit = currentDrawdown + targetProfit;
   
   double stepDistance = GetCurrentStepDistance();
   double pipValue = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
   double pipSize = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
   double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
   
   double stepPips = stepDistance * (point / pipSize);
   
   if(stepPips > 0 && pipValue > 0)
   {
      double lot = (requiredProfit / (stepPips * pipValue)) * RecoveryFactor;
      return lot;
   }
   
   return CalculateLotSize(totalOrders + 1);
}

//+------------------------------------------------------------------+
//| Update Order Information                                          |
//+------------------------------------------------------------------+
void UpdateOrderInfo()
{
   totalOrders = 0;
   totalBuyOrders = 0;
   totalSellOrders = 0;
   totalBuyLots = 0;
   totalSellLots = 0;
   totalLots = 0;
   totalProfit = 0;
   double sumPrice = 0;
   lastPrice = 0;
   
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol)
      {
         if(PositionGetInteger(POSITION_MAGIC) == MagicNumber)
         {
            totalOrders++;
            double lots = PositionGetDouble(POSITION_VOLUME);
            double openPrice = PositionGetDouble(POSITION_PRICE_OPEN);
            double profit = PositionGetDouble(POSITION_PROFIT);
            double swap = PositionGetDouble(POSITION_SWAP);
            double commission = PositionGetDouble(POSITION_COMMISSION);
            ENUM_POSITION_TYPE posType = (ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE);
            
            totalLots += lots;
            sumPrice += openPrice * lots;
            totalProfit += profit + swap + commission;
            
            if(posType == POSITION_TYPE_BUY)
            {
               totalBuyOrders++;
               totalBuyLots += lots;
            }
            else
            {
               totalSellOrders++;
               totalSellLots += lots;
            }
            
            if(totalOrders == 1) 
            {
               lastPrice = openPrice;
            }
         }
      }
   }
   
   if(totalLots > 0)
      averagePrice = sumPrice / totalLots;
   else
      averagePrice = 0;
}

//+------------------------------------------------------------------+
//| Check RSI and Price Action                                        |
//+------------------------------------------------------------------+
int CheckRSI_PA()
{
   datetime currentBarTime = iTime(_Symbol, PERIOD_CURRENT, 0);
   if(currentBarTime == lastBarTime) return 0;
   
   double rsi[];
   ArraySetAsSeries(rsi, true);
   if(CopyBuffer(rsiHandle, 0, 0, 2, rsi) < 2)
      return 0;
   
   double open = iOpen(_Symbol, PERIOD_CURRENT, 1);
   double close = iClose(_Symbol, PERIOD_CURRENT, 1);
   double high = iHigh(_Symbol, PERIOD_CURRENT, 1);
   double low = iLow(_Symbol, PERIOD_CURRENT, 1);
   
   double candleSize = high - low;
   double bodySize = MathAbs(close - open);
   double bodyPercent = (bodySize / candleSize) * 100;
   
   if(rsi[1] < RSI_Oversold)
   {
      if(RequireBullishCandle)
      {
         if(close > open && bodyPercent >= MinCandleBodyPercent)
            return 1;
      }
      else
      {
         return 1;
      }
   }
   
   if(rsi[1] > RSI_Overbought)
   {
      if(RequireBearishCandle)
      {
         if(close < open && bodyPercent >= MinCandleBodyPercent)
            return -1;
      }
      else
      {
         return -1;
      }
   }
   
   return 0;
}

//+------------------------------------------------------------------+
//| Open Order By RSI + PA                                            |
//+------------------------------------------------------------------+
void OpenOrderByRSI_PA()
{
   int signal = CheckRSI_PA();
   if(signal == 0) return;
   
   MqlTradeRequest request = {};
   MqlTradeResult result = {};
   
   double price;
   ENUM_ORDER_TYPE type;
   
   if(signal == 1)
   {
      price = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      type = ORDER_TYPE_BUY;
   }
   else
   {
      price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      type = ORDER_TYPE_SELL;
   }
   
   double lot = CalculateLotSize(1);
   
   request.action = TRADE_ACTION_DEAL;
   request.symbol = _Symbol;
   request.volume = lot;
   request.type = type;
   request.price = price;
   request.deviation = Slippage;
   request.magic = MagicNumber;
   request.comment = TradeComment + "_RSI_PA_1";
   
   if(OrderSend(request, result))
   {
      if(result.retcode == TRADE_RETCODE_DONE || result.retcode == TRADE_RETCODE_PLACED)
      {
         Print("RSI+PA Order Opened: Type=", EnumToString(type), ", Lot=", lot, ", Price=", price);
         lastBarTime = iTime(_Symbol, PERIOD_CURRENT, 0);
      }
   }
}

//+------------------------------------------------------------------+
//| Check and Open Next Order By RSI PA                               |
//+------------------------------------------------------------------+
void CheckAndOpenNextOrderByRSI_PA()
{
   if(totalOrders == 0) return;
   
   ENUM_POSITION_TYPE lastType;
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol)
      {
         if(PositionGetInteger(POSITION_MAGIC) == MagicNumber)
         {
            lastPrice = PositionGetDouble(POSITION_PRICE_OPEN);
            lastType = (ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE);
            break;
         }
      }
   }
   
   double currentPrice;
   double distance;
   
   if(lastType == POSITION_TYPE_BUY)
   {
      currentPrice = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      distance = lastPrice - currentPrice;
   }
   else
   {
      currentPrice = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      distance = currentPrice - lastPrice;
   }
   
   double stepDistance = GetCurrentStepDistance();
   
   if(distance >= stepDistance)
   {
      ENUM_ORDER_TYPE typeToOpen = (lastType == POSITION_TYPE_BUY) ? ORDER_TYPE_BUY : ORDER_TYPE_SELL;
      double lot = CalculateRecoveryLot(typeToOpen);
      OpenNextOrder(lot, typeToOpen);
   }
}

//+------------------------------------------------------------------+
//| Open First Order (Grid Mode)                                      |
//+------------------------------------------------------------------+
void OpenFirstOrder()
{
   MqlTradeRequest request = {};
   MqlTradeResult result = {};
   
   double price;
   ENUM_ORDER_TYPE type;
   
   if(GridDirection == ORDER_TYPE_BUY)
   {
      price = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      type = ORDER_TYPE_BUY;
   }
   else
   {
      price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      type = ORDER_TYPE_SELL;
   }
   
   double lot = CalculateLotSize(1);
   
   request.action = TRADE_ACTION_DEAL;
   request.symbol = _Symbol;
   request.volume = lot;
   request.type = type;
   request.price = price;
   request.deviation = Slippage;
   request.magic = MagicNumber;
   request.comment = TradeComment + "_1";
   
   if(OrderSend(request, result))
   {

   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol)
      {

         {
            lastPrice = PositionGetDouble(POSITION_PRICE_OPEN);
            lastType = (ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE);
            break;
         }
      }
   }
   
   double currentPrice;
   double distance;
   
   if(lastType == POSITION_TYPE_BUY)
   {
      currentPrice = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      distance = lastPrice - currentPrice;
   }
   else
   {
      currentPrice = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      distance = currentPrice - lastPrice;
   }
   
   double stepDistance = GetCurrentStepDistance();
   
   if(distance >= stepDistance)
   {
      ENUM_ORDER_TYPE typeToOpen = (lastType == POSITION_TYPE_BUY) ? ORDER_TYPE_BUY : ORDER_TYPE_SELL;
      double lot = CalculateRecoveryLot(typeToOpen);
      OpenNextOrder(lot, typeToOpen);
   }
}

void CloseWorstOrder()
{
   double worstProfit = 0;
   ulong worstTicket = 0;
   double worstVolume = 0;
   ENUM_POSITION_TYPE worstType;
   
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {

      {
         if(PositionGetInteger(POSITION_MAGIC) == MagicNumber)
         {

            if(profit < worstProfit || worstTicket == 0)
            {
               worstProfit = profit;
               worstTicket = PositionGetInteger(POSITION_TICKET);
               worstVolume = PositionGetDouble(POSITION_VOLUME);
               worstType = (ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE);
            }
         }
      }
   }
   
   if(worstTicket > 0)
   {
      MqlTradeRequest request = {};
      MqlTradeResult result = {};
      
      request.action = TRADE_ACTION_DEAL;
      request.position = worstTicket;
      request.symbol = _Symbol;
      request.volume = worstVolume;
      request.deviation = Slippage;
      request.magic = MagicNumber;
      
      if(worstType == POSITION_TYPE_BUY)
      {
         request.type = ORDER_TYPE_SELL;
         request.price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      }
      else
      {
         request.type = ORDER_TYPE_BUY;
         request.price = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      }
      
      if(OrderSend(request, result))
      {
         if(result.retcode == TRADE_RETCODE_DONE || result.retcode == TRADE_RETCODE_PLACED)
         {
            Print("*** PARTIAL CLOSE: Closed worst order #", worstTicket, " with loss $", worstProfit, " ***");
         }
      }
   }
}

//| Normalize Lot Size                                                |
//+------------------------------------------------------------------+
double NormalizeLot(double lot)
{
   double minLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
   double maxLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MAX);
   double lotStep = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
   
   if(lot < minLot) lot = minLot;
   if(lot > maxLot) lot = maxLot;
   
   lot = MathFloor(lot / lotStep) * lotStep;
   
   return lot;
}

## 💰 2. MONEY MANAGEMENT - การจัดการเงิน

### 2.1 LOT MODE - วิธีคำนวณ Lot

#### แบบ FIXED (Lot คงที่)
```
LotMode = LOT_FIXED
FirstLot = 0.01
LotMultiplier = 1.5
```

**การทำงาน:**
- ไม้ที่ 1: 0.01 lot
- ไม้ที่ 2: 0.01 × 1.5 = 0.015 lot
- ไม้ที่ 3: 0.015 × 1.5 = 0.0225 lot
- ไม้ที่ 4: 0.0225 × 1.5 = 0.03375 lot

**สูตร:** `Lot = FirstLot × (Multiplier ^ (n-1))`

**เหมาะกับ:** เทรดเดอร์ที่ต้องการความเสี่ยงคงที่

```
Balance $5,000:
- FirstLot = 5000/1000 × 1.0 = 0.05 lot
- ไม้ที่ 2: 0.05 × 1.5 = 0.075 lot
- ไม้ที่ 3: 0.075 × 1.5 = 0.1125 lot

Balance $10,000:
- FirstLot = 10000/1000 × 1.0 = 0.10 lot
- ไม้ที่ 2: 0.10 × 1.5 = 0.15 lot
```

**สูตร:** `Lot = (Balance / 1000) × PercentBalance × (Multiplier ^ (n-1))`

**✅ ข้อดี:**
- ปรับตาม account size
- Compound เงินอัตโนมัติ
- Risk management แบบ %

**เหมาะกับ:** เทรดเดอร์ที่ต้องการ scale ตามเงิน

ATR_Period = 14
```

**ATR คืออะไร?**
- Average True Range
- วัดความผันผวนของราคา
- ตลาดผันผวน = ATR สูง
- ตลาดนิ่ง = ATR ต่ำ

**การทำงาน:**
```
สมมติ ATR(H1) = 50 points
ATR_Multiplier = 1.5

แบบไม่คูณ (ATR_MultiplyByOrder = false):
ทุกไม้: 50 × 1.5 = 75 points

แบบคูณ (ATR_MultiplyByOrder = true, GrowthFactor = 1.0):
ไม้ที่ 1: 50 × 1.5 × 1 = 75 points
ไม้ที่ 2: 50 × 1.5 × 2 = 150 points
ไม้ที่ 3: 50 × 1.5 × 3 = 225 points
ไม้ที่ 4: 50 × 1.5 × 4 = 300 points
```

**สูตร:** `Distance = ATR × Multiplier × (OrderNumber ^ GrowthFactor)`

**ตัวอย่างในสถานการณ์ต่างๆ:**

**ตลาดนิ่ง (Low Volatility):**
```
ATR = 30 points
ไม้ที่ 1: 30 × 1.5 × 1 = 45 points
ไม้ที่ 2: 30 × 1.5 × 2 = 90 points
ไม้ที่ 3: 30 × 1.5 × 3 = 135 points
```

**ตลาดผันผวน (High Volatility):**
```
ATR = 80 points
ไม้ที่ 1: 80 × 1.5 × 1 = 120 points
ไม้ที่ 2: 80 × 1.5 × 2 = 240 points
ไม้ที่ 3: 80 × 1.5 × 3 = 360 points
```

**✅ ข้อดี:**
- ปรับตามสภาพตลาด AUTO
- ตลาดนิ่ง = ไม้ถี่ (จับ reversal)
- ตลาดแรง = ไม้ห่าง (ลดความเสี่ยง)
- Professional มาก

**⚠️ ข้อเสีย:**
- ซับซ้อนที่สุด
- ต้องเข้าใจ ATR
- Grid distance เปลี่ยนตลอด

**เหมาะกับ:** มือโปร, ตลาดผันผวนบ่อย

## 🔄 5. RECOVERY LOT MODE - ระบบแก้ไม้ (สำคัญ!)

### 5.1 AW FORMULA (Original)
```
RecoveryLotMode = RECOVERY_AW_FORMULA
RecoveryFactor = 1.3
```

**สูตร:**
```
RecoveryLot = (CurrentLoss + TargetProfit) / (StepDistance × PipValue) × Factor
```

**ตัวอย่าง:**
```
สถานการณ์:
- ขาดทุน: -$100
- เป้ากำไร: $50
- Step Distance: 30 pips
- Pip Value: $10 per lot
- Factor: 1.3

RecoveryLot = (100 + 50) / (30 × 10) × 1.3
           = 150 / 300 × 1.3
           = 0.65 lot

→ เปิดไม้ถัดไป 0.65 lot
   ถ้าราคากลับ 30 pips = กำไร ~$195
   Cover Loss $100 + Profit $50 = ✅
```

**✅ ข้อดี:**
- Recover เร็วมาก
- คำนวณแม่นยำ
- มีเป้าหมายชัดเจน

**⚠️ ข้อเสีย:**
- Lot เพิ่มเร็วมากใน deep drawdown
- ต้อง Factor ที่เหมาะสม (1.2-1.5)

**Factor Guide:**
- 1.2 = Conservative (ปลอดภัย)
- 1.3 = Moderate (ปานกลาง)
- 1.5 = Aggressive (เสี่ยง)

---

### 5.2 DRAWDOWN BASED (ตามจำนวนเงินขาดทุน)
```
RecoveryLotMode = RECOVERY_DRAWDOWN_BASED
DrawdownPerLot = 100  // $100 loss = +0.01 lot
```

**การทำงาน:**
```
ขาดทุน $0 → 0.01 lot (FirstLot)
ขาดทุน $100 → 0.02 lot
ขาดทุน $200 → 0.03 lot
ขาดทุน $500 → 0.06 lot
ขาดทุน $1000 → 0.11 lot
```

**สูตร:** `Lot = (Loss / DrawdownPerLot) × 0.01`

**✅ ข้อดี:**
- เข้าใจง่าย
- ปรับตามขาดทุนจริง
- คำนวณง่าย

**⚠️ ข้อเสีย:**
- อาจเร็วเกินไปใน trending
- ไม่คำนึงถึงขนาด account

---

### 5.3 PERCENT DRAWDOWN (ตาม % ขาดทุน)
```
RecoveryLotMode = RECOVERY_PERCENT_DRAWDOWN
DrawdownPercentPerLot = 1.0  // 1% = +0.01 lot
```

**การทำงาน:**
```
Balance $5,000:
- Drawdown 1% ($50) → 0.02 lot
- Drawdown 5% ($250) → 0.06 lot
- Drawdown 10% ($500) → 0.11 lot

Balance $10,000:
- Drawdown 1% ($100) → 0.02 lot
- Drawdown 5% ($500) → 0.06 lot
- Drawdown 10% ($1000) → 0.11 lot
```

**สูตร:** `Lot = (DrawdownPercent / DrawdownPercentPerLot) × 0.01`

**✅ ข้อดี:**
- ปรับตาม account size
- Risk management เป็น %
- เหมาะทุก account

---

### 5.4 OPPOSITE PERCENT (Hedge ทิศทางตรงข้าม) 🌟 แนะนำ!
```
RecoveryLotMode = RECOVERY_OPPOSITE_PERCENT
OppositePercentOfTotal = 30  // 30%
EnableOppositeHedge = true
AutoResetOnWrong = true
ResetLossThreshold = -50
```

**การทำงาน:**

**Scenario 1: เปิด BUY แล้วราคาลง**
```
Entry: BUY 0.10 lot ที่ 1.1000
ราคาลง 1.0970 → ขาดทุน -$30

Recovery:
- Total BUY lots = 0.10
- เปิด SELL = 0.10 × 30% = 0.03 lot

ผลลัพธ์:
- ถ้าราคาลงต่อ: SELL กำไร, BUY ขาดทุนช้าลง
- ถ้าราคาขึ้น: BUY กำไร, SELL ขาดทุนเล็กน้อย
- Net Risk ลดลง! 🛡️
```

**Scenario 2: Hedge มากขึ้น**
```
Position:
- BUY 0.10 lot
- BUY 0.15 lot
- BUY 0.20 lot
Total BUY = 0.45 lot

ราคายังลง, ขาดทุน -$150

Recovery:
- เปิด SELL = 0.45 × 30% = 0.135 lot

ผล:
- SELL hedge ส่วนหนึ่งของ BUY
- ลด net exposure จาก 0.45 → 0.315
```

**Auto Reset:**
```
ถ้า Total Loss > -$50:
→ ปิดทั้งหมด
→ Reset
→ รอสัญญาณใหม่
```

**✅ ข้อดี:**
- 🛡️ ลดความเสี่ยงมาก
- 💰 กำไรได้ทั้ง 2 ทิศทาง
- 🔄 Auto reset ป้องกัน blow up
- 🎯 เหมาะตลาดไม่แน่นอน
- ✅ ปลอดภัยที่สุด

**⚠️ ข้อเสีย:**
- พลาดกำไรเต็มจาก strong trend
- Spread/Commission เยอะขึ้น
- ต้องเป็น netting account (MT5)

**OppositePercent Guide:**
- 20% = Conservative (hedge น้อย)
- 30% = Moderate (แนะนำ)
- 40% = Aggressive (hedge มาก)
- 50% = Full hedge (เสมอกัน)

# EA AW Recover V4.2 - Grid Hedge + Dynamic RSI

แก้ไขตามที่ขอแล้วครับ เพิ่ม **Hedge Grid** และ **Dynamic RSI**!

```mql5
//+------------------------------------------------------------------+
//|                              AW_Recover_Grid_Dynamic_RSI.mq5     |
//|                                   Quant Trader - Forex Recovery  |
//+------------------------------------------------------------------+
#property copyright "Quant Trader"
#property version   "4.20"
#property strict

//--- Enums
enum ENUM_ENTRY_MODE
{
   ENTRY_GRID = 0,              // Grid (One Direction)
   ENTRY_GRID_HEDGE = 1,        // Grid Hedge (Both Directions)
   ENTRY_FOLLOW_TREND = 2,      // Follow Trend (Candle Color)
   ENTRY_RSI_PA = 3             // RSI + Price Action
};

enum ENUM_GRID_TYPE
{
   GRID_FIXED = 0,              // Fixed Distance
   GRID_LINEAR = 1,             // Linear Increase (Distance × Order Number)
   GRID_DYNAMIC_ATR = 2         // Dynamic ATR (ATR × Order Number)
};

enum ENUM_TP_MODE
{
   TP_FIXED_PIP = 0,            // Fixed Pips
   TP_PERCENT_BALANCE = 1       // Percent of Balance
};

enum ENUM_LOT_MODE
{
   LOT_FIXED = 0,               // Fixed Lot
   LOT_PERCENT_BALANCE = 1      // Percent of Balance
};

enum ENUM_RECOVERY_LOT_MODE
{
   RECOVERY_AW_FORMULA = 0,          // AW Formula (Original)
   RECOVERY_DRAWDOWN_BASED = 1,      // Based on Drawdown ($)
   RECOVERY_PERCENT_DRAWDOWN = 2,    // Based on Drawdown (%)
   RECOVERY_OPPOSITE_PERCENT = 3,    // Opposite Direction % of Total Lots
   RECOVERY_MARTINGALE = 4           // Simple Martingale
};

enum ENUM_RSI_MODE
{
   RSI_FIXED_LEVEL = 0,         // Fixed Level (30/70)
   RSI_DYNAMIC = 1              // Dynamic (Based on History)
};

//--- Input Parameters
input group "===== Main Entry Mode ====="
input ENUM_ENTRY_MODE EntryMode = ENTRY_RSI_PA;        // Entry Mode
input bool EnableRecoveryMode = true;                   // Enable Recovery Mode
input ENUM_ORDER_TYPE GridDirection = ORDER_TYPE_BUY;  // Grid Direction (for One Direction Grid)

input group "===== GRID HEDGE SETTINGS ====="
input double HedgeGridStartPrice = 0;                  // Hedge Grid Start Price (0=Current Price)

//+------------------------------------------------------------------+
//| Expert tick function                                              |
//+------------------------------------------------------------------+
void OnTick()
{
   //--- Update order info
   UpdateOrderInfo();
   
   //--- Check for reset condition
   if(AutoResetOnWrong && needsReset)
   {
      if(totalProfit <= ResetLossThreshold)
      {
         Print("*** AUTO RESET: Closing all positions. Loss: $", totalProfit, " ***");
         CloseAllOrders();
         needsReset = false;
         isRecoveryMode = false;
         return;
      }
   }
   
   //--- Check drawdown percentage
   double accountBalance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
   double currentDrawdownPercent = (totalProfit / accountBalance) * 100;
   
   if(currentDrawdownPercent < maxDrawdown)
      maxDrawdown = currentDrawdownPercent;
   
   //--- Check if we should enter recovery mode
   if(EnableRecoveryMode && !isRecoveryMode && totalOrders == 0)
   {
      if(MathAbs(currentDrawdownPercent) >= DrawdownPercentToStart)
      {
         isRecoveryMode = true;
         Print("*** RECOVERY MODE ACTIVATED at ", MathAbs(currentDrawdownPercent), "% Drawdown ***");
      }
   }
   
   //--- Check if we need to close all orders (target reached)
   if(totalOrders > 0)
   {
      double targetProfit = CalculateTargetProfit();
      
      if(totalProfit >= targetProfit)
      {
         CloseAllOrders();
         Print("*** TARGET PROFIT REACHED: $", totalProfit, " (Target: $", targetProfit, ") ***");
         isRecoveryMode = false;
         maxDrawdown = 0;
         needsReset = false;
         return;
      }
      
      //--- Check max drawdown
      if(MaxDrawdownMoney > 0 && totalProfit <= -MaxDrawdownMoney)
      {
         CloseAllOrders();
         Print("*** MAX DRAWDOWN (Money) REACHED: $", totalProfit, " ***");
         isRecoveryMode = false;
         maxDrawdown = 0;
         needsReset = false;
         return;
      }
      
      if(MaxDrawdownPercent > 0 && MathAbs(currentDrawdownPercent) >= MaxDrawdownPercent)
      {
         CloseAllOrders();
         Print("*** MAX DRAWDOWN (Percent) REACHED: ", MathAbs(currentDrawdownPercent), "% ***");
         isRecoveryMode = false;
         maxDrawdown = 0;
         needsReset = false;
         return;
      }
   }
   
   //--- Entry logic based on mode
   if(EntryMode == ENTRY_GRID)
   {
      if(totalOrders == 0)
      {
         OpenFirstOrder();
         return;
      }
      
      if(totalOrders > 0 && totalOrders < MaxOrders)
      {
         CheckAndOpenNextOrder();
      }
   }
   else if(EntryMode == ENTRY_GRID_HEDGE)
   {
      //--- Grid Hedge Mode
      if(totalOrders == 0)
      {
         OpenHedgeGridFirstOrders();
         return;
      }
      
      if(totalOrders > 0 && totalOrders < MaxOrders * 2) // *2 because both directions
      {
         CheckAndOpenNextHedgeOrders();
      }
   }
   else if(EntryMode == ENTRY_FOLLOW_TREND)
   {
      if(totalOrders == 0)
      {
         OpenFirstOrderByTrend();
         return;
      }
      
      if(totalOrders > 0 && totalOrders < MaxOrders)
      {
         CheckAndOpenNextOrderByTrend();
      }
   }
   else if(EntryMode == ENTRY_RSI_PA)
   {
      if(totalOrders == 0 || (!EnableRecoveryMode))
      {
         OpenOrderByRSI_PA();
         return;
      }
      
      if(EnableRecoveryMode && totalOrders > 0 && totalOrders < MaxOrders)
      {
         CheckAndOpenNextOrderByRSI_PA();
      }
   }
}

//+------------------------------------------------------------------+
//| Calculate Dynamic RSI Levels                                      |
//+------------------------------------------------------------------+
void CalculateDynamicRSI(double &oversold, double &overbought)
{
   double rsiArray[];
   ArraySetAsSeries(rsiArray, true);
   
   if(CopyBuffer(rsiHandle, 0, 1, RSI_LookbackBars, rsiArray) < RSI_LookbackBars)
   {
      // Fallback to fixed levels
      oversold = RSI_Oversold;
      overbought = RSI_Overbought;
      return;
   }
   
   //--- Find min and max RSI
   double minRSI = rsiArray[ArrayMinimum(rsiArray)];
   double maxRSI = rsiArray[ArrayMaximum(rsiArray)];
   double rangeRSI = maxRSI - minRSI;
   
   //--- Calculate threshold based on percentile
   double threshold = rangeRSI * (RSI_PercentileThreshold / 100.0);
   
   oversold = minRSI + threshold;
   overbought = maxRSI - threshold;
   
   if(ShowGridInfo)
   {
      Print("📊 Dynamic RSI: Min=", minRSI, ", Max=", maxRSI, 
            ", Oversold=", oversold, ", Overbought=", overbought);
   }
}

//+------------------------------------------------------------------+
//| Check RSI and Price Action (Improved with Dynamic RSI)            |
//+------------------------------------------------------------------+
int CheckRSI_PA()
{
   datetime currentBarTime = iTime(_Symbol, PERIOD_CURRENT, 0);
   if(currentBarTime == lastBarTime) return 0;
   
   double rsi[];
   ArraySetAsSeries(rsi, true);
   if(CopyBuffer(rsiHandle, 0, 0, 2, rsi) < 2)
      return 0;
   
   //--- Get dynamic or fixed RSI levels
   double oversoldLevel, overboughtLevel;
   
   if(RSI_Mode == RSI_DYNAMIC)
   {
      CalculateDynamicRSI(oversoldLevel, overboughtLevel);
   }
   else
   {
      oversoldLevel = RSI_Oversold;
      overboughtLevel = RSI_Overbought;
   }
   
   //--- Get candle data
   double open = iOpen(_Symbol, PERIOD_CURRENT, 1);
   double close = iClose(_Symbol, PERIOD_CURRENT, 1);
   double high = iHigh(_Symbol, PERIOD_CURRENT, 1);
   double low = iLow(_Symbol, PERIOD_CURRENT, 1);
   
   double candleSize = high - low;
   if(candleSize == 0) return 0;
   
   double bodySize = MathAbs(close - open);
   double bodyPercent = (bodySize / candleSize) * 100;
   
   //--- Check for BUY signal
   if(rsi[1] < oversoldLevel)
   {
      if(RequireBullishCandle)
      {
         if(close > open && bodyPercent >= MinCandleBodyPercent)
         {
            Print("✅ BUY Signal: RSI=", rsi[1], " < ", oversoldLevel, ", Bullish Candle");
            return 1;
         }
      }
      else
      {
         Print("✅ BUY Signal: RSI=", rsi[1], " < ", oversoldLevel);
         return 1;
      }
   }
   
   //--- Check for SELL signal
   if(rsi[1] > overboughtLevel)
   {
      if(RequireBearishCandle)
      {
         if(close < open && bodyPercent >= MinCandleBodyPercent)
         {
            Print("✅ SELL Signal: RSI=", rsi[1], " > ", overboughtLevel, ", Bearish Candle");
            return -1;
         }
      }
      else
      {
         Print("✅ SELL Signal: RSI=", rsi[1], " > ", overboughtLevel);
         return -1;
      }
   }
   
   return 0;
}

//| Open Hedge Grid First Orders                                      |

void OpenHedgeGridFirstOrders()
{

      }
   }
   
   Sleep(200);
   
   //--- Open SELL
   double sellPrice = SymbolInfoDouble(_Symbol, SYMBOL_BID);

   }
}

//+------------------------------------------------------------------+
//| Check and Open Next Hedge Orders                                  |
//+------------------------------------------------------------------+
void CheckAndOpenNextHedgeOrders()
{
   if(totalOrders == 0) return;
   
   double currentAsk = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
   double currentBid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
   double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
   
   //--- Check BUY side
   if(totalBuyOrders < MaxOrders && lastBuyPrice > 0)
   {
      double buyDistance = lastBuyPrice - currentAsk;
      int buyDistancePoints = (int)(buyDistance / point);
      
      if(buyDistancePoints >= HedgeGridBuyDistance)
      {
         double buyLot = CalculateRecoveryLot(ORDER_TYPE_BUY);
         OpenNextOrder(buyLot, ORDER_TYPE_BUY);
      }
   }
   
   //--- Check SELL side
   if(totalSellOrders < MaxOrders && lastSellPrice > 0)
   {
      double sellDistance = currentBid - lastSellPrice;
      int sellDistancePoints = (int)(sellDistance / point);
      
      if(sellDistancePoints >= HedgeGridSellDistance)
      {
         double sellLot = CalculateRecoveryLot(ORDER_TYPE_SELL);
         OpenNextOrder(sellLot, ORDER_TYPE_SELL);
      }
   }
}

//+------------------------------------------------------------------+
//| Calculate Target Profit                                           |
//+------------------------------------------------------------------+
double CalculateTargetProfit()
{
   double target = 0;
   
   if(TakeProfitMode == TP_FIXED_PIP)
   {
      double pipValue = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
      double pipSize = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
      double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
      
      double pipsInPoints = TakeProfitPips * (pipSize / point);
      target = pipsInPoints * pipValue * totalLots;
   }
   else if(TakeProfitMode == TP_PERCENT_BALANCE)
   {
      double balance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
      target = balance * (TakeProfitPercent / 100.0);
   }
   
   return target;
}

//+------------------------------------------------------------------+
//| Calculate Lot Size (Main Entry)                                   |
//+------------------------------------------------------------------+
double CalculateLotSize(int orderNumber = 1)
{
   double lot = 0;
   
   if(LotMode == LOT_FIXED)
   {
      lot = FirstLot * MathPow(LotMultiplier, orderNumber - 1);
   }
   else if(LotMode == LOT_PERCENT_BALANCE)
   {
      double balance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
      lot = (balance / 1000.0) * LotPercentBalance;
      
      if(orderNumber > 1)
         lot = lot * MathPow(LotMultiplier, orderNumber - 1);
   }
   
   if(lot > MaxLotSize)
      lot = MaxLotSize;
   
   return NormalizeLot(lot);
}

//+------------------------------------------------------------------+
//| Calculate Recovery Lot (Advanced Methods)                         |
//+------------------------------------------------------------------+
double CalculateRecoveryLot(ENUM_ORDER_TYPE desiredType)
{
   if(!EnableRecoveryMode || totalOrders == 0)
      return CalculateLotSize(totalOrders + 1);
   
   double lot = 0;
   double balance = AccountBalanceForCalc > 0 ? AccountBalanceForCalc : AccountInfoDouble(ACCOUNT_BALANCE);
   double currentDrawdown = MathAbs(totalProfit);
   double currentDrawdownPercent = (totalProfit / balance) * 100;
   
   switch(RecoveryLotMode)
   {
      case RECOVERY_AW_FORMULA:
         lot = CalculateAWLot();
         break;
         
      case RECOVERY_DRAWDOWN_BASED:
         lot = (currentDrawdown / DrawdownPerLot) * 0.01;
         lot = MathMax(lot, FirstLot);
         break;
         
      case RECOVERY_PERCENT_DRAWDOWN:
         lot = (MathAbs(currentDrawdownPercent) / DrawdownPercentPerLot) * 0.01;
         lot = MathMax(lot, FirstLot);
         break;
         
      case RECOVERY_OPPOSITE_PERCENT:
         if(EnableOppositeHedge)
         {
            if(desiredType == ORDER_TYPE_BUY && totalSellLots > totalBuyLots)
            {
               lot = totalSellLots * (OppositePercentOfTotal / 100.0);
            }
            else if(desiredType == ORDER_TYPE_SELL && totalBuyLots > totalSellLots)
            {
               lot = totalBuyLots * (OppositePercentOfTotal / 100.0);
            }
            else
            {
               lot = CalculateLotSize(totalOrders + 1);
            }
            
            if(totalProfit < 0)
               needsReset = true;
         }
         else
         {
            lot = CalculateLotSize(totalOrders + 1);
         }
         break;
         
      case RECOVERY_MARTINGALE:
         lot = CalculateLotSize(totalOrders + 1);
         break;
   }
   
   if(lot > MaxLotSize)
      lot = MaxLotSize;
   
   return NormalizeLot(lot);
}

//+------------------------------------------------------------------+
//| Calculate AW Recovery Lot                                         |
//+------------------------------------------------------------------+
double CalculateAWLot()
{
   if(totalProfit >= 0)
      return CalculateLotSize(totalOrders + 1);
   
   double currentDrawdown = MathAbs(totalProfit);
   double targetProfit = CalculateTargetProfit();
   double requiredProfit = currentDrawdown + targetProfit;
   
   double stepDistance = GetCurrentStepDistance();
   double pipValue = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
   double pipSize = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
   double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
   
   double stepPips = stepDistance * (point / pipSize);
   
   if(stepPips > 0 && pipValue > 0)
   {
      double lot = (requiredProfit / (stepPips * pipValue)) * RecoveryFactor;
      return lot;
   }
   
   return CalculateLotSize(totalOrders + 1);
}

//+------------------------------------------------------------------+
//| Get Current Step Distance                                         |
//+------------------------------------------------------------------+
double GetCurrentStepDistance()
{
   double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
   int nextOrderNumber = totalOrders + 1;
   
   if(GridType == GRID_FIXED)
   {
      return FixedStepPoints * point;
   }
   else if(GridType == GRID_LINEAR)
   {
      if(LinearMultiplyByOrder)
      {
         double multiplier = MathPow(nextOrderNumber, LinearGrowthFactor);
         return LinearBaseStep * multiplier * point;
      }
      else
      {
         return LinearBaseStep * point;
      }
   }
   else if(GridType == GRID_DYNAMIC_ATR)
   {
      double atr[];
      ArraySetAsSeries(atr, true);
      
      if(CopyBuffer(atrHandle, 0, 0, 1, atr) > 0)
      {
         if(ATR_MultiplyByOrder)
         {
            double multiplier = MathPow(nextOrderNumber, ATR_GrowthFactor);
            return atr[0] * ATR_Multiplier * multiplier;
         }
         else
         {
            return atr[0] * ATR_Multiplier;
         }
      }
      
      return FixedStepPoints * point;
   }
   
   return FixedStepPoints * point;
}

//+------------------------------------------------------------------+
//| Update Order Information                                          |
//+------------------------------------------------------------------+
void UpdateOrderInfo()
{
   totalOrders = 0;
   totalBuyOrders = 0;
   totalSellOrders = 0;
   totalBuyLots = 0;
   totalSellLots = 0;
   totalLots = 0;
   totalProfit = 0;
   double sumPrice = 0;
   lastBuyPrice = 0;
   lastSellPrice = 0;
   
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol)
      {
         if(PositionGetInteger(POSITION_MAGIC) == MagicNumber)
         {
            totalOrders++;
            double lots = PositionGetDouble(POSITION_VOLUME);
            double openPrice = PositionGetDouble(POSITION_PRICE_OPEN);
            double profit = PositionGetDouble(POSITION_PROFIT);
            double swap = PositionGetDouble(POSITION_SWAP);
            double commission = PositionGetDouble(POSITION_COMMISSION);
            ENUM_POSITION_TYPE posType = (ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE);
            
            totalLots += lots;
            sumPrice += openPrice * lots;
            totalProfit += profit + swap + commission;
            
            if(posType == POSITION_TYPE_BUY)
            {
               totalBuyOrders++;
               totalBuyLots += lots;
               if(lastBuyPrice == 0 || openPrice < lastBuyPrice)
                  lastBuyPrice = openPrice;
            }
            else
            {
               totalSellOrders++;
               totalSellLots += lots;
               if(lastSellPrice == 0 || openPrice > lastSellPrice)
                  lastSellPrice = openPrice;
            }
         }
      }
   }
   
   if(totalLots > 0)
      averagePrice = sumPrice / totalLots;
   else
      averagePrice = 0;
}

//| Open Order By RSI + PA                                            |
//+------------------------------------------------------------------+

{
   int signal = CheckRSI_PA();
   if(signal == 0) return;
   
   MqlTradeRequest request = {};
   MqlTradeResult result = {};
   
   double price;
   ENUM_ORDER_TYPE type;
   
   if(signal == 1)
   {
      price = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      type = ORDER_TYPE_BUY;
   }
   else
   {
      price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      type = ORDER_TYPE_SELL;
   }
   
   double lot = CalculateLotSize(1);
   
   request.action = TRADE_ACTION_DEAL;
   request.symbol = _Symbol;
   request.volume = lot;
   request.type = type;
   request.price = price;
   request.deviation = Slippage;
   request.magic = MagicNumber;
   request.comment = TradeComment + "_RSI_PA_1";
   
   if(OrderSend(request, result))
   {
      if(result.retcode == TRADE_RETCODE_DONE || result.retcode == TRADE_RETCODE_PLACED)
      {
         Print("RSI+PA Order Opened: Type=", EnumToString(type), ", Lot=", lot, ", Price=", price);
         lastBarTime = iTime(_Symbol, PERIOD_CURRENT, 0);
      }
   }
}

//+------------------------------------------------------------------+
//| Check and Open Next Order By RSI PA                               |
//+------------------------------------------------------------------+
void CheckAndOpenNextOrderByRSI_PA()
{
   if(totalOrders == 0) return;
   
   ENUM_POSITION_TYPE lastType;

//| Close All Orders                                                  |

void CloseAllOrders()
{
   MqlTradeRequest request = {};
   MqlTradeResult result = {};
   
   int closed = 0;
   
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol)
      {
         if(PositionGetInteger(POSITION_MAGIC) == MagicNumber)
         {
            ulong ticket = PositionGetInteger(POSITION_TICKET);
            double volume = PositionGetDouble(POSITION_VOLUME);
            ENUM_POSITION_TYPE posType = (ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE);
            
            request.action = TRADE_ACTION_DEAL;
            request.position = ticket;
            request.symbol = _Symbol;
            request.volume = volume;
            request.deviation = Slippage;
            request.magic = MagicNumber;
            
            if(posType == POSITION_TYPE_BUY)
            {
               request.type = ORDER_TYPE_SELL;
               request.price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
            }
            else
            {
               request.type = ORDER_TYPE_BUY;
               request.price = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
            }
            
            if(OrderSend(request, result))
            {
               if(result.retcode == TRADE_RETCODE_DONE || result.retcode == TRADE_RETCODE_PLACED)
               {
                  closed++;
               }
            }
            
            Sleep(100);
         }
      }
   }
   
   Print("=== Closed ", closed, " orders. Final Profit: $", DoubleToString(totalProfit, 2), " ===");
}

//+------------------------------------------------------------------+
//| Normalize Lot Size                                                |
//+------------------------------------------------------------------+
double NormalizeLot(double lot)
{
   double minLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
   double maxLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MAX);
   double lotStep = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
   
   if(lot < minLot) lot = minLot;
   if(lot > maxLot) lot = maxLot;
   
   lot = MathFloor(lot / lotStep) * lotStep;
   
   return lot;
}

//+------------------------------------------------------------------+
//| Expert tick function                                              |
//+------------------------------------------------------------------+
void OnTick()
{
   UpdateOrderInfo();
   
   //--- Check for auto reset
   if(Hedge_AutoReset && needsReset && totalProfit <= Hedge_ResetLoss)
   {
      Print("🔄 AUTO RESET: Loss = $", totalProfit);
      CloseAllOrders();
      needsReset = false;
      isRecoveryMode = false;
      lastTPTime = 0;
      waitBarsCount = 0;
      return;
   }
   
   //--- Check drawdown
   double balance = AccountInfoDouble(ACCOUNT_BALANCE);
   double ddPercent = (totalProfit / balance) * 100;
   
   //--- Safety checks
   if(Safety_MaxDD_Money > 0 && totalProfit <= -Safety_MaxDD_Money)
   {
      Print("🛑 MAX DRAWDOWN ($): Loss = $", totalProfit);
      CloseAllOrders();
      return;
   }
   
   if(Safety_MaxDD_Percent > 0 && MathAbs(ddPercent) >= Safety_MaxDD_Percent)
   {
      Print("🛑 MAX DRAWDOWN (%): ", MathAbs(ddPercent), "%");
      CloseAllOrders();
      return;
   }
   
   //--- Start recovery mode
   if(EnableRecovery && !isRecoveryMode && totalOrders == 0)
   {
      if(MathAbs(ddPercent) >= Safety_StartRecovery)
      {
         isRecoveryMode = true;
         Print("⚡ RECOVERY MODE ON: Drawdown = ", MathAbs(ddPercent), "%");
      }
   }
   
   //--- Check TP
   if(totalOrders > 0)
   {
      double target = CalculateTargetProfit();
      
      if(totalProfit >= target)
      {
         Print("🎯 TARGET REACHED: $", totalProfit, " (Target: $", target, ")");
         CloseAllOrders();
         lastTPTime = TimeCurrent();
         waitBarsCount = 0;
         isRecoveryMode = false;
         needsReset = false;
         return;
      }
   }
   
   //--- Smart Entry: Wait after TP
   if(UseSmartEntry && lastTPTime > 0 && totalOrders == 0)
   {
      datetime currentBar = iTime(_Symbol, PERIOD_CURRENT, 0);
      
      if(currentBar > lastBarTime)
      {
         waitBarsCount++;
         lastBarTime = currentBar;
         
         if(waitBarsCount <= WaitBarsAfterTP)
         {
            if(ShowInfo)
               Print("⏳ Waiting... Bar ", waitBarsCount, "/", WaitBarsAfterTP);
            return;
         }
      }
      else
      {
         return; // Still same bar
      }
   }
   
   //--- Entry Logic
   if(totalOrders == 0)
   {
      if(UseSmartEntry && lastTPTime > 0)
      {
         // Smart entry: analyze direction
         int direction = AnalyzeDirection();
         
         if(direction == 0)
         {
            if(ShowInfo)
               Print("🤔 Not confident - Waiting...");
            return;
         }
         
         OpenSmartOrder(direction);
         lastTPTime = 0;
         waitBarsCount = 0;
      }
      else
      {
         // Normal entry
         if(EntryMode == ENTRY_GRID)
            OpenFirstOrder();
         else if(EntryMode == ENTRY_GRID_HEDGE)
            OpenHedgeGridFirstOrders();
         else if(EntryMode == ENTRY_RSI_PA)
            OpenOrderByRSI_PA();
      }
   }
   else if(totalOrders < MaxOrders)
   {
      // Add more orders
      if(EntryMode == ENTRY_GRID)
         CheckAndOpenNextOrder();
      else if(EntryMode == ENTRY_GRID_HEDGE)
         CheckAndOpenNextHedgeOrders();
      else if(EntryMode == ENTRY_RSI_PA)
         CheckAndOpenNextOrderByRSI_PA();
   }
}

//+------------------------------------------------------------------+
//| Analyze Direction (for Smart Entry)                              |
//+------------------------------------------------------------------+
int AnalyzeDirection()
{
   if(DirectionFilter == DIRECTION_NONE)
      return GridDirection == ORDER_TYPE_BUY ? 1 : -1;
   
   if(DirectionFilter == DIRECTION_RSI)
   {
      double rsi[];
      ArraySetAsSeries(rsi, true);
      if(CopyBuffer(rsiHandle, 0, 0, 2, rsi) < 2)
         return 0;
      
      double oversold, overbought;
      
      if(RSI_Mode == 2) // Dynamic
         CalculateDynamicRSI(oversold, overbought);
      else
      {
         oversold = RSI_Oversold;
         overbought = RSI_Overbought;
      }
      
      if(rsi[1] < oversold)
      {
         Print("📊 RSI Analysis: ", rsi[1], " < ", oversold, " → BUY");
         return 1;
      }
      else if(rsi[1] > overbought)
      {
         Print("📊 RSI Analysis: ", rsi[1], " > ", overbought, " → SELL");
         return -1;
      }
      else
      {
         Print("📊 RSI Analysis: ", rsi[1], " → Neutral (Waiting)");
         return 0;
      }
   }
   else if(DirectionFilter == DIRECTION_MA)
   {
      double ma[], close[];
      ArraySetAsSeries(ma, true);
      ArraySetAsSeries(close, true);
      
      if(CopyBuffer(maHandle, 0, 0, 1, ma) < 1)
         return 0;
      if(CopyClose(_Symbol, PERIOD_CURRENT, 0, 1, close) < 1)
         return 0;
      
      if(close[0] > ma[0])
      {
         Print("📈 MA Analysis: Price > MA → BUY");
         return 1;
      }
      else if(close[0] < ma[0])
      {
         Print("📉 MA Analysis: Price < MA → SELL");
         return -1;
      }
      else
      {
         return 0;
      }
   }
   else if(DirectionFilter == DIRECTION_PRICE_ACTION)
   {
      double open = iOpen(_Symbol, PERIOD_CURRENT, 1);
      double close = iClose(_Symbol, PERIOD_CURRENT, 1);
      double high = iHigh(_Symbol, PERIOD_CURRENT, 1);
      double low = iLow(_Symbol, PERIOD_CURRENT, 1);
      
      double body = MathAbs(close - open);
      double total = high - low;
      
      if(total == 0) return 0;
      
      double bodyPercent = (body / total) * 100;
      
      if(close > open && bodyPercent >= MinBodyPercent)
      {
         Print("🕯️ PA Analysis: Bullish Candle (", bodyPercent, "%) → BUY");
         return 1;
      }
      else if(close < open && bodyPercent >= MinBodyPercent)
      {
         Print("🕯️ PA Analysis: Bearish Candle (", bodyPercent, "%) → SELL");
         return -1;
      }
      else
      {
         Print("🕯️ PA Analysis: Weak candle → Waiting");
         return 0;
      }
   }
   
   return 0;
}

//+------------------------------------------------------------------+
//| Calculate Dynamic RSI                                             |
//+------------------------------------------------------------------+
void CalculateDynamicRSI(double &oversold, double &overbought)
{
   double rsiArray[];
   ArraySetAsSeries(rsiArray, true);
   
   if(CopyBuffer(rsiHandle, 0, 1, RSI_Lookback, rsiArray) < RSI_Lookback)
   {
      oversold = RSI_Oversold;
      overbought = RSI_Overbought;
      return;
   }
   
   double minRSI = rsiArray[ArrayMinimum(rsiArray)];
   double maxRSI = rsiArray[ArrayMaximum(rsiArray)];
   double rangeRSI = maxRSI - minRSI;
   double threshold = rangeRSI * (RSI_Threshold / 100.0);
   
   oversold = minRSI + threshold;
   overbought = maxRSI - threshold;
}

//+------------------------------------------------------------------+
//| Calculate Target Profit                                           |
//+------------------------------------------------------------------+
double CalculateTargetProfit()
{
   if(TP_Mode == 1)
   {
      double pipValue = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
      double pipSize = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
      double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
      double pipsInPoints = TP_Pips * (pipSize / point);
      return pipsInPoints * pipValue * (totalBuyLots + totalSellLots);
   }
   else
   {
      double balance = AccountInfoDouble(ACCOUNT_BALANCE);
      return balance * (TP_Percent / 100.0);
   }
}

//+------------------------------------------------------------------+
//| Calculate Lot                                                     |
//+------------------------------------------------------------------+
double CalculateLotSize(int orderNumber)
{
   double lot = Lot_Start * MathPow(Lot_Multiplier, orderNumber - 1);
   if(lot > Lot_Max) lot = Lot_Max;
   return NormalizeLot(lot);
}

//+------------------------------------------------------------------+
//| Calculate Recovery Lot                                            |
//+------------------------------------------------------------------+
double CalculateRecoveryLot(ENUM_ORDER_TYPE type)
{
   if(!EnableRecovery || totalOrders == 0)
      return CalculateLotSize(totalOrders + 1);
   
   double lot = 0;
   double balance = AccountInfoDouble(ACCOUNT_BALANCE);
   double dd = MathAbs(totalProfit);
   
   switch(Recovery_Mode)
   {
      case 1: // AW Formula
         if(totalProfit < 0)
         {
            double target = CalculateTargetProfit();
            double required = dd + target;
            double stepDist = GetStepDistance();
            double pipValue = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
            double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
            
            if(stepDist > 0 && pipValue > 0)
            {
               double stepPips = stepDist / point;
               lot = (required / (stepPips * pipValue)) * AW_Factor;
            }
         }
         if(lot == 0) lot = CalculateLotSize(totalOrders + 1);
         break;
         
      case 2: // Drawdown Based
         lot = (dd / DD_PerLot) * 0.01;
         if(lot < Lot_Start) lot = Lot_Start;
         break;
         
      case 3: // Hedge
         if(type == ORDER_TYPE_BUY && totalSellLots > totalBuyLots)
            lot = totalSellLots * (Hedge_Percent / 100.0);
         else if(type == ORDER_TYPE_SELL && totalBuyLots > totalSellLots)
            lot = totalBuyLots * (Hedge_Percent / 100.0);
         else
            lot = CalculateLotSize(totalOrders + 1);
         
         if(totalProfit < 0) needsReset = true;
         break;
         
      case 4: // Martingale
         lot = CalculateLotSize(totalOrders + 1);
         break;
   }
   
   if(lot > Lot_Max) lot = Lot_Max;
   return NormalizeLot(lot);
}

//+------------------------------------------------------------------+
//| Get Step Distance                                                 |
//+------------------------------------------------------------------+
double GetStepDistance()
{
   double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
   int nextOrder = totalOrders + 1;
   
   if(Distance_Mode == 1)
   {
      return Distance_Fixed * point;
   }
   else if(Distance_Mode == 2)
   {
      if(Distance_MultiplyByOrder)
         return Distance_Base * nextOrder * point;
      else
         return Distance_Base * point;
   }
   else if(Distance_Mode == 3)
   {
      double atr[];
      ArraySetAsSeries(atr, true);
      if(CopyBuffer(atrHandle, 0, 0, 1, atr) > 0)
      {
         if(ATR_MultiplyByOrder)
            return atr[0] * ATR_Multiplier * nextOrder;
         else
            return atr[0] * ATR_Multiplier;
      }
      return Distance_Fixed * point;
   }
   
   return Distance_Fixed * point;
}

//+------------------------------------------------------------------+
//| Update Order Info                                                 |
//+------------------------------------------------------------------+
void UpdateOrderInfo()
{
   totalOrders = 0;
   totalBuyOrders = 0;
   totalSellOrders = 0;
   totalBuyLots = 0;
   totalSellLots = 0;
   totalProfit = 0;
   lastBuyPrice = 0;
   lastSellPrice = 0;
   
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol && PositionGetInteger(POSITION_MAGIC) == MagicNumber)
      {
         totalOrders++;
         double lots = PositionGetDouble(POSITION_VOLUME);
         double price = PositionGetDouble(POSITION_PRICE_OPEN);
         totalProfit += PositionGetDouble(POSITION_PROFIT) + PositionGetDouble(POSITION_SWAP) + PositionGetDouble(POSITION_COMMISSION);
         
         if(PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_BUY)
         {
            totalBuyOrders++;
            totalBuyLots += lots;
            if(lastBuyPrice == 0 || price < lastBuyPrice) lastBuyPrice = price;
         }
         else
         {
            totalSellOrders++;
            totalSellLots += lots;
            if(lastSellPrice == 0 || price > lastSellPrice) lastSellPrice = price;
         }
      }
   }
}

//+------------------------------------------------------------------+
//| Check and Open Next Order                                         |
//+------------------------------------------------------------------+
void CheckAndOpenNextOrder()
{
   if(totalOrders == 0) return;
   
   double currentPrice = (GridDirection == ORDER_TYPE_BUY) ? SymbolInfoDouble(_Symbol, SYMBOL_ASK) : SymbolInfoDouble(_Symbol, SYMBOL_BID);
   double lastPrice = (GridDirection == ORDER_TYPE_BUY) ? lastBuyPrice : lastSellPrice;
   
   if(lastPrice == 0) return;
   
   double distance = (GridDirection == ORDER_TYPE_BUY) ? (lastPrice - currentPrice) : (currentPrice - lastPrice);
   double stepDist = GetStepDistance();
   
   if(distance >= stepDist)
   {
      MqlTradeRequest request = {};
      MqlTradeResult result = {};
      
      request.action = TRADE_ACTION_DEAL;
      request.symbol = _Symbol;
      request.volume = CalculateRecoveryLot(GridDirection);
      request.type = GridDirection;
      request.price = currentPrice;
      request.deviation = 30;
      request.magic = MagicNumber;
      request.comment = TradeComment + "_" + IntegerToString(totalOrders + 1);
      
      OrderSend(request, result);
   }
}

//| Check and Open Next Hedge Orders                                  |
//+------------------------------------------------------------------+
void CheckAndOpenNextHedgeOrders()
{
   double stepDist = GetStepDistance();
   
   // BUY side
   if(totalBuyOrders < MaxOrders && lastBuyPrice > 0)
   {
      double currentAsk = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      if((lastBuyPrice - currentAsk) >= stepDist)
      {
         MqlTradeRequest request = {};
         MqlTradeResult result = {};
         
         request.action = TRADE_ACTION_DEAL;
         request.symbol = _Symbol;
         request.volume = CalculateRecoveryLot(ORDER_TYPE_BUY);
         request.type = ORDER_TYPE_BUY;
         request.price = currentAsk;
         request.deviation = 30;
         request.magic = MagicNumber;
         request.comment = TradeComment + "_HedgeBUY";
         
         OrderSend(request, result);
      }
   }
   
   // SELL side
   if(totalSellOrders < MaxOrders && lastSellPrice > 0)
   {
      double currentBid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      if((currentBid - lastSellPrice) >= stepDist)
      {
         MqlTradeRequest request = {};
         MqlTradeResult result = {};
         
         request.action = TRADE_ACTION_DEAL;
         request.symbol = _Symbol;
         request.volume = CalculateRecoveryLot(ORDER_TYPE_SELL);
         request.type = ORDER_TYPE_SELL;
         request.price = currentBid;
         request.deviation = 30;
         request.magic = MagicNumber;
         request.comment = TradeComment + "_HedgeSELL";
         
         OrderSend(request, result);
      }
   }
}

//+------------------------------------------------------------------+
//| Check RSI PA                                                      |
//+------------------------------------------------------------------+
int CheckRSI_PA()
{
   datetime currentBar = iTime(_Symbol, PERIOD_CURRENT, 0);
   if(currentBar == lastBarTime) return 0;
   
   double rsi[];
   ArraySetAsSeries(rsi, true);
   if(CopyBuffer(rsiHandle, 0, 0, 2, rsi) < 2) return 0;
   
   double oversold, overbought;
   if(RSI_Mode == 2)
      CalculateDynamicRSI(oversold, overbought);
   else
   {
      oversold = RSI_Oversold;
      overbought = RSI_Overbought;
   }
   
   double open = iOpen(_Symbol, PERIOD_CURRENT, 1);
   double close = iClose(_Symbol, PERIOD_CURRENT, 1);
   double high = iHigh(_Symbol, PERIOD_CURRENT, 1);
   double low = iLow(_Symbol, PERIOD_CURRENT, 1);
   
   double body = MathAbs(close - open);
   double total = high - low;
   if(total == 0) return 0;
   
   double bodyPercent = (body / total) * 100;
   
   if(rsi[1] < oversold)
   {
      if(!Require_PA || (close > open && bodyPercent >= MinBodyPercent))
         return 1;
   }
   
   if(rsi[1] > overbought)
   {
      if(!Require_PA || (close < open && bodyPercent >= MinBodyPercent))
         return -1;
   }
   
   return 0;
}

//+------------------------------------------------------------------+
//| Open Order By RSI PA                                              |
//+------------------------------------------------------------------+
void OpenOrderByRSI_PA()
{
   int signal = CheckRSI_PA();
   if(signal == 0) return;
   
   MqlTradeRequest request = {};
   MqlTradeResult result = {};
   
   request.action = TRADE_ACTION_DEAL;
   request.symbol = _Symbol;
   request.volume = NormalizeLot(Lot_Start);
   request.deviation = 30;
   request.magic = MagicNumber;
   request.comment = TradeComment + "_RSI";
   
   if(signal == 1)
   {
      request.type = ORDER_TYPE_BUY;
      request.price = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
   }
   else
   {
      request.type = ORDER_TYPE_SELL;
      request.price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
   }
   
   OrderSend(request, result);
   lastBarTime = iTime(_Symbol, PERIOD_CURRENT, 0);
}

//+------------------------------------------------------------------+
//| Check and Open Next Order By RSI PA                               |
//+------------------------------------------------------------------+
void CheckAndOpenNextOrderByRSI_PA()
{
   if(totalOrders == 0) return;
   
   double lastPrice = 0;
   ENUM_POSITION_TYPE lastType;
   
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol && PositionGetInteger(POSITION_MAGIC) == MagicNumber)
      {
         lastPrice = PositionGetDouble(POSITION_PRICE_OPEN);
         lastType = (ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE);
         break;
      }
   }
   
   if(lastPrice == 0) return;
   
   double currentPrice = (lastType == POSITION_TYPE_BUY) ? SymbolInfoDouble(_Symbol, SYMBOL_ASK) : SymbolInfoDouble(_Symbol, SYMBOL_BID);
   double distance = (lastType == POSITION_TYPE_BUY) ? (lastPrice - currentPrice) : (currentPrice - lastPrice);
   
   if(distance >= GetStepDistance())
   {
      MqlTradeRequest request = {};
      MqlTradeResult result = {};
      
      request.action = TRADE_ACTION_DEAL;
      request.symbol = _Symbol;
      request.volume = CalculateRecoveryLot(lastType == POSITION_TYPE_BUY ? ORDER_TYPE_BUY : ORDER_TYPE_SELL);
      request.type = lastType == POSITION_TYPE_BUY ? ORDER_TYPE_BUY : ORDER_TYPE_SELL;
      request.price = currentPrice;
      request.deviation = 30;
      request.magic = MagicNumber;
      request.comment = TradeComment + "_" + IntegerToString(totalOrders + 1);
      
      OrderSend(request, result);
   }
}

//+------------------------------------------------------------------+
//| Close All Orders                                                  |
//+------------------------------------------------------------------+
void CloseAllOrders()
{
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol && PositionGetInteger(POSITION_MAGIC) == MagicNumber)
      {
         MqlTradeRequest request = {};
         MqlTradeResult result = {};
         
         request.action = TRADE_ACTION_DEAL;
         request.position = PositionGetInteger(POSITION_TICKET);
         request.symbol = _Symbol;
         request.volume = PositionGetDouble(POSITION_VOLUME);
         request.deviation = 30;
         request.magic = MagicNumber;
         
         if(PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_BUY)
         {
            request.type = ORDER_TYPE_SELL;
            request.price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
         }
         else
         {
            request.type = ORDER_TYPE_BUY;
            request.price = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
         }
         
         OrderSend(request, result);
         Sleep(100);
      }
   }
   
   Print("✅ All orders closed. Final P/L: $", DoubleToString(totalProfit, 2));
}

//+------------------------------------------------------------------+
//| Normalize Lot                                                     |
//+------------------------------------------------------------------+
double NormalizeLot(double lot)
{
   double minLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
   double maxLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MAX);
   double lotStep = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
   
   if(lot < minLot) lot = minLot;
   if(lot > maxLot) lot = maxLot;
   
   return MathFloor(lot / lotStep) * lotStep;
}

### 🆕 **1.1 Smart Entry (ใหม่!)**

```

**ทำอะไร:**
- หลังจาก TP แล้ว → รอ 1 แท่ง
- วิเคราะห์ว่าควรเปิด BUY หรือ SELL
- ถ้าไม่แน่ใจ → รอต่อ

**ตัวเลือก DirectionFilter:**

| Filter | ค่า | คำอธิบาย |
|--------|-----|---------|
| **[0] None** | `DIRECTION_NONE` | เปิดตามที่ตั้งไว้ทันที |
| **[1] RSI** | `DIRECTION_RSI` | ใช้ RSI ตัดสิน (แนะนำ!) |
| **[2] MA** | `DIRECTION_MA` | ใช้ MA ตัดสิน |
| **[3] PA** | `DIRECTION_PRICE_ACTION` | ใช้แท่งเทียนตัดสิน |

**ตัวอย่าง:**
```
TP ที่ $50 → ปิดทั้งหมด
รอ 1 แท่งเทียน...
ตรวจ RSI:
- RSI = 32 (Oversold) → เปิด BUY ✅
- RSI = 50 (Neutral) → รอต่อ ⏳
- RSI = 72 (Overbought) → เปิด SELL ✅
```
